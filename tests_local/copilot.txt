from __future__ import annotations

import copy
import json
import logging
from typing import Any

from playwright.async_api import Page

from q_webgis.configuration import (
    WEBGIS_BAADPLADS_UPDATE_URL,
    WEBGIS_TIMEOUT_MS,
)


logger = logging.getLogger(__name__)


# ------------------------------------------------------------
# Verificerede driftstatuskoder
# ------------------------------------------------------------

DRIFTSTATUS_FRI = "1"
DRIFTSTATUS_UDLEJET = "3"

KENDTE_DRIFTSTATUS_LABELS = {
    DRIFTSTATUS_FRI: "Fri",
    DRIFTSTATUS_UDLEJET: "Udlejet",
}


# ------------------------------------------------------------
# START: aendr_baadplads_status
# ------------------------------------------------------------

async def aendr_baadplads_status(
    page: Page,
    ticket: str,
    baadplads: dict[str, Any],
    ny_driftstatus: int | str,
    *,
    timeout_ms: int = WEBGIS_TIMEOUT_MS,
) -> dict[str, Any]:
    """
    Ændrer driftstatus på én angivet bådplads.

    Funktionen opdaterer kun den bådplads, der sendes ind i
    parameteren baadplads. Havn, strækning, plads og ogc_fid
    læses fra inputobjektet og er ikke hardkodet.

    Funktionen accepterer både:

        1. Et komplet SFS-featureobjekt med attributes.

        2. En flad bådpladsrække, hvis rækken indeholder wkt,
           projection, hasattachments, ogc_fid og driftstatus.

    Parametre:
        page:
            Den åbne og autentificerede WebGIS-side.

        ticket:
            Ticket-headeren opsnappet fra WebGIS' egne requests.

        baadplads:
            Den bådplads, der skal ændres.

        ny_driftstatus:
            Den ønskede tekniske statuskode.

            Verificerede værdier:

                1 = Fri
                3 = Udlejet

        timeout_ms:
            Maksimal ventetid for PUT-requestet.

    Output:
        Dictionary med identifikation, tidligere status, ny status,
        HTTP-status og WebGIS-svar.

    Sikkerhed:
        Ticket, cookies og credentials bliver ikke logget.
    """
    _valider_page(
        page
    )

    valideret_ticket = _valider_tekst(
        value=ticket,
        field_name="ticket",
    )

    valideret_timeout_ms = _valider_positivt_heltal(
        value=timeout_ms,
        field_name="timeout_ms",
    )

    valideret_driftstatus = _valider_driftstatus(
        ny_driftstatus
    )

    original_feature = _normaliser_baadplads_feature(
        baadplads
    )

    feature_til_opdatering = copy.deepcopy(
        original_feature
    )

    feature_til_opdatering[
        "attributes"
    ][
        "driftstatus"
    ] = valideret_driftstatus

    oprindelige_attributes = original_feature[
        "attributes"
    ]

    attributes = feature_til_opdatering[
        "attributes"
    ]

    ogc_fid = attributes[
        "ogc_fid"
    ]

    tidligere_driftstatus = str(
        oprindelige_attributes.get(
            "driftstatus",
            "",
        )
    ).strip()

    havn = _hent_valgfri_attribut(
        attributes=attributes,
        field_name="havn",
    )

    straekning = _hent_valgfri_attribut(
        attributes=attributes,
        field_name="straekning",
    )

    plads = _hent_valgfri_attribut(
        attributes=attributes,
        field_name="plads",
    )

    status_label = KENDTE_DRIFTSTATUS_LABELS[
        valideret_driftstatus
    ]

    if tidligere_driftstatus == valideret_driftstatus:
        logger.info(
            "Bådpladsen har allerede den ønskede status. "
            "ogc_fid=%s, havn=%r, plads=%r, status=%s (%s).",
            ogc_fid,
            havn,
            plads,
            valideret_driftstatus,
            status_label,
        )

        return {
            "success": True,
            "opdateret": False,
            "oversprunget": True,
            "aarsag": (
                "Bådpladsen har allerede den ønskede status."
            ),
            "ogc_fid": ogc_fid,
            "havn": havn,
            "straekning": straekning,
            "plads": plads,
            "tidligere_driftstatus": tidligere_driftstatus,
            "ny_driftstatus": valideret_driftstatus,
            "status_label": status_label,
            "http_status": None,
            "response_data": None,
        }

    logger.info(
        "Ændrer bådpladsstatus. ogc_fid=%s, havn=%r, "
        "strækning=%r, plads=%r, fra=%r, til=%r (%s).",
        ogc_fid,
        havn,
        straekning,
        plads,
        tidligere_driftstatus,
        valideret_driftstatus,
        status_label,
    )

    request_payload = {
        "features": [
            feature_til_opdatering
        ]
    }

    browser_resultat = await page.evaluate(
        """
        async ({
            url,
            ticket,
            requestPayload,
            timeoutMs
        }) => {
            const controller = new AbortController();

            const timeoutHandle = setTimeout(
                () => controller.abort(),
                timeoutMs
            );

            try {
                const response = await fetch(
                    url,
                    {
                        method: "PUT",
                        credentials: "include",
                        headers: {
                            "Accept": (
                                "application/json, "
                                + "text/javascript, "
                                + "*/*; q=0.01"
                            ),
                            "Content-Type": (
                                "application/json; "
                                + "charset=UTF-8"
                            ),
                            "Ticket": ticket,
                            "X-Requested-With": (
                                "XMLHttpRequest"
                            )
                        },
                        body: JSON.stringify(
                            requestPayload
                        ),
                        signal: controller.signal
                    }
                );

                const responseText = (
                    await response.text()
                );

                return {
                    ok: response.ok,
                    status: response.status,
                    statusText: response.statusText,
                    contentType: (
                        response.headers.get(
                            "content-type"
                        )
                        || ""
                    ),
                    responseText: responseText,
                    finalUrl: response.url,
                    browserError: ""
                };
            } catch (error) {
                return {
                    ok: false,
                    status: 0,
                    statusText: "",
                    contentType: "",
                    responseText: "",
                    finalUrl: "",
                    browserError: (
                        error instanceof Error
                            ? (
                                error.name
                                + ": "
                                + error.message
                            )
                            : String(error)
                    )
                };
            } finally {
                clearTimeout(
                    timeoutHandle
                );
            }
        }
        """,
        {
            "url": WEBGIS_BAADPLADS_UPDATE_URL,
            "ticket": valideret_ticket,
            "requestPayload": request_payload,
            "timeoutMs": valideret_timeout_ms,
        },
    )

    fetch_resultat = _valider_fetch_resultat(
        browser_resultat
    )

    browser_error = str(
        fetch_resultat.get("browserError")
        or ""
    ).strip()

    if browser_error:
        raise RuntimeError(
            "WebGIS-kaldet til ændring af bådpladsstatus "
            "kunne ikke udføres. "
            f"ogc_fid={ogc_fid}. "
            f"Browserfejl: {browser_error}"
        )

    status = fetch_resultat[
        "status"
    ]

    status_text = str(
        fetch_resultat.get("statusText")
        or ""
    ).strip()

    content_type = str(
        fetch_resultat.get("contentType")
        or ""
    ).strip()

    response_text = str(
        fetch_resultat.get("responseText")
        or ""
    ).strip()

    final_url = str(
        fetch_resultat.get("finalUrl")
        or ""
    ).strip()

    if not fetch_resultat[
        "ok"
    ]:
        raise RuntimeError(
            "WebGIS afviste ændringen af bådpladsstatus. "
            f"ogc_fid={ogc_fid}. "
            f"HTTP-status={status}. "
            f"Statusbesked={status_text!r}. "
            f"Content-Type={content_type!r}. "
            f"Slut-URL={final_url!r}. "
            "Svareksempel: "
            f"{_lav_sikkert_response_preview(response_text)}"
        )

    response_data = _parse_response(
        response_text=response_text,
        content_type=content_type,
    )

    logger.info(
        "Bådpladsstatus er ændret. ogc_fid=%s, "
        "havn=%r, plads=%r, ny status=%s (%s).",
        ogc_fid,
        havn,
        plads,
        valideret_driftstatus,
        status_label,
    )

    return {
        "success": True,
        "opdateret": True,
        "oversprunget": False,
        "aarsag": None,
        "ogc_fid": ogc_fid,
        "havn": havn,
        "straekning": straekning,
        "plads": plads,
        "tidligere_driftstatus": tidligere_driftstatus,
        "ny_driftstatus": valideret_driftstatus,
        "status_label": status_label,
        "http_status": status,
        "response_data": response_data,
    }

# ------------------------------------------------------------
# Feature-normalisering
# ------------------------------------------------------------

def _normaliser_baadplads_feature(
    baadplads: dict[str, Any],
) -> dict[str, Any]:
    """
    Normaliserer input til det komplette SFS-featureformat.

    Et allerede komplet featureobjekt kopieres uændret.
    En flad række pakkes ind i attributes.
    """
    if not isinstance(
        baadplads,
        dict,
    ):
        raise TypeError(
            "baadplads skal være en dictionary."
        )

    if isinstance(
        baadplads.get("attributes"),
        dict,
    ):
        feature = copy.deepcopy(
            baadplads
        )
    else:
        envelope_fields = {
            "wkt",
            "projection",
            "hasattachments",
        }

        attributes = {
            key: copy.deepcopy(value)
            for key, value in baadplads.items()
            if key not in envelope_fields
        }

        feature = {
            "wkt": baadplads.get("wkt"),
            "projection": (
                baadplads.get("projection")
                or "EPSG:25832"
            ),
            "hasattachments": baadplads.get(
                "hasattachments",
                False,
            ),
            "attributes": attributes,
        }

    _valider_baadplads_feature(
        feature
    )

    return feature


def _valider_baadplads_feature(
    feature: dict[str, Any],
) -> None:
    """Validerer felterne, som det verificerede PUT kræver."""
    wkt = feature.get(
        "wkt"
    )

    if (
        not isinstance(wkt, str)
        or not wkt.strip()
    ):
        raise ValueError(
            "Bådpladsen mangler wkt. Der skal bruges "
            "det komplette featureobjekt fra WebGIS."
        )

    projection = feature.get(
        "projection"
    )

    if (
        not isinstance(projection, str)
        or not projection.strip()
    ):
        raise ValueError(
            "Bådpladsen mangler projection."
        )

    if not isinstance(
        feature.get("hasattachments"),
        bool,
    ):
        raise TypeError(
            "hasattachments skal være boolsk."
        )

    attributes = feature.get(
        "attributes"
    )

    if not isinstance(
        attributes,
        dict,
    ):
        raise TypeError(
            "Bådpladsen mangler attributes."
        )

    if attributes.get("ogc_fid") is None:
        raise ValueError(
            "Bådpladsen mangler ogc_fid."
        )

    if "driftstatus" not in attributes:
        raise ValueError(
            "Bådpladsen mangler driftstatus."
        )


def _hent_valgfri_attribut(
    *,
    attributes: dict[str, Any],
    field_name: str,
) -> str | None:
    """Returnerer en valgfri attribut som trimmet tekst."""
    value = attributes.get(
        field_name
    )

    if value is None:
        return None

    text = str(
        value
    ).strip()

    return text or None


# ------------------------------------------------------------
# Validering
# ------------------------------------------------------------

def _valider_driftstatus(
    value: Any,
) -> str:
    """Validerer de statuskoder, der er verificeret i WebGIS."""
    if isinstance(
        value,
        bool,
    ):
        raise TypeError(
            "ny_driftstatus må ikke være boolsk."
        )

    if not isinstance(
        value,
        (
            int,
            str,
        ),
    ):
        raise TypeError(
            "ny_driftstatus skal være et heltal "
            "eller tekst."
        )

    driftstatus = str(
        value
    ).strip()

    if driftstatus not in KENDTE_DRIFTSTATUS_LABELS:
        raise ValueError(
            "Kun statuskode 1 (Fri) og 3 (Udlejet) "
            "er verificeret."
        )

    return driftstatus


def _valider_fetch_resultat(
    value: Any,
) -> dict[str, Any]:
    """Validerer resultatet fra browserens fetch-kald."""
    if not isinstance(
        value,
        dict,
    ):
        raise RuntimeError(
            "WebGIS-statuskaldet returnerede et "
            "ugyldigt browserresultat."
        )

    status = value.get(
        "status"
    )

    if (
        isinstance(status, bool)
        or not isinstance(status, int)
    ):
        raise RuntimeError(
            "WebGIS-statuskaldet returnerede en "
            "ugyldig HTTP-status."
        )

    if not isinstance(
        value.get("ok"),
        bool,
    ):
        raise RuntimeError(
            "WebGIS-statuskaldet returnerede ikke "
            "en gyldig ok-status."
        )

    return value


def _valider_page(
    page: Page,
) -> None:
    """Kontrollerer, at Playwright-siden kan anvendes."""
    if page is None:
        raise ValueError(
            "page skal angives."
        )

    if page.is_closed():
        raise RuntimeError(
            "WebGIS-siden er lukket og kan ikke "
            "bruges til at ændre bådpladsstatus."
        )


def _valider_tekst(
    *,
    value: Any,
    field_name: str,
) -> str:
    """Returnerer en obligatorisk, trimmet tekstværdi."""
    if not isinstance(
        value,
        str,
    ):
        raise TypeError(
            f"{field_name} skal være tekst."
        )

    cleaned_value = value.strip()

    if not cleaned_value:
        raise ValueError(
            f"{field_name} må ikke være tom."
        )

    return cleaned_value


def _valider_positivt_heltal(
    *,
    value: Any,
    field_name: str,
) -> int:
    """Returnerer værdien som et positivt heltal."""
    if isinstance(
        value,
        bool,
    ):
        raise TypeError(
            f"{field_name} må ikke være boolsk."
        )

    try:
        integer_value = int(
            value
        )
    except (
        TypeError,
        ValueError,
    ) as exc:
        raise TypeError(
            f"{field_name} skal være et heltal."
        ) from exc

    if integer_value <= 0:
        raise ValueError(
            f"{field_name} skal være større end 0."
        )

    return integer_value


# ------------------------------------------------------------
# Response
# ------------------------------------------------------------

def _parse_response(
    *,
    response_text: str,
    content_type: str,
) -> Any:
    """Parser WebGIS-svaret som JSON, tekst eller None."""
    if not response_text:
        return None

    ligner_json = (
        "json" in content_type.casefold()
        or response_text.lstrip().startswith("{")
        or response_text.lstrip().startswith("[")
    )

    if not ligner_json:
        return response_text

    try:
        return json.loads(
            response_text
        )
    except json.JSONDecodeError as exc:
        raise RuntimeError(
            "WebGIS returnerede et svar, der ligner "
            "JSON, men ikke kunne parses. "
            f"JSON-fejl: {exc.msg}."
        ) from exc


def _lav_sikkert_response_preview(
    value: Any,
    maximum_length: int = 500,
) -> str:
    """Laver et begrænset preview af WebGIS-svaret."""
    text = " ".join(
        str(
            value
            or ""
        ).split()
    )

    if not text:
        return "<tomt svar>"

    if len(text) <= maximum_length:
        return text

    return (
        text[:maximum_length]
        + "..."
    )

from __future__ import annotations

import logging
from typing import Any

from playwright.async_api import Page

from q_webgis.configuration import (
    WEBGIS_MAXROWS,
    WEBGIS_TIMEOUT_MS,
)
from q_webgis.functionality.spatialanalyze import (
    hent_spatialanalyze_raekker,
)


logger = logging.getLogger(__name__)


# ------------------------------------------------------------
# Bådpladsindstillinger
# ------------------------------------------------------------

BAADPLADS_MAXROWS = WEBGIS_MAXROWS
BAADPLADS_JSONTYPES = "true"
BAADPLADS_OUTPUTFORMAT = "json"
BAADPLADS_ERROR_CONTENT_TYPE = "json"


# ------------------------------------------------------------
# START: hent_baadpladser
# ------------------------------------------------------------

async def hent_baadpladser(
    page: Page,
    session_id: str,
    *,
    maxrows: int = BAADPLADS_MAXROWS,
    timeout_ms: int = WEBGIS_TIMEOUT_MS,
) -> list[dict[str, Any]]:
    """
    Henter alle bådpladser fra den seneste spatialanalyse i den
    aktuelle WebGIS-session.

    Funktionen starter ikke selv spatialanalysen. Analysen for
    ds-baadplads skal være gennemført først med samme Page,
    JSESSIONID og Session-Id.

    Parametre:
        page:
            Den åbne og autentificerede WebGIS-side.

        session_id:
            Session-Id opsnappet fra WebGIS' egne requests.

        maxrows:
            Maksimalt antal rækker, WebGIS må returnere.

        timeout_ms:
            Maksimal ventetid for requestet i millisekunder.

    Output:
        En liste med alle returnerede bådpladsrækker.

        Hver række returneres uændret fra den eksisterende
        hent_spatialanalyze_raekker()-funktion.

    Fejl:
        RuntimeError hvis WebGIS-siden er lukket, eller hvis
        resultatet rammer maxrows og derfor kan være afkortet.
    """
    _valider_page(
        page
    )

    valideret_session_id = _valider_tekst(
        value=session_id,
        field_name="session_id",
    )

    valideret_maxrows = _valider_positivt_heltal(
        value=maxrows,
        field_name="maxrows",
    )

    valideret_timeout_ms = _valider_positivt_heltal(
        value=timeout_ms,
        field_name="timeout_ms",
    )

    logger.info(
        "Henter op til %s bådpladser fra den seneste "
        "WebGIS-spatialanalyse.",
        valideret_maxrows,
    )

    baadpladser = await hent_spatialanalyze_raekker(
        page=page,
        session_id=valideret_session_id,
        maxrows=valideret_maxrows,
        jsontypes=BAADPLADS_JSONTYPES,
        outputformat=BAADPLADS_OUTPUTFORMAT,
        error_content_type=(
            BAADPLADS_ERROR_CONTENT_TYPE
        ),
        timeout_ms=valideret_timeout_ms,
    )

    _valider_baadpladsraekker(
        baadpladser
    )

    if len(baadpladser) >= valideret_maxrows:
        raise RuntimeError(
            "WebGIS returnerede det maksimalt tilladte "
            "antal bådpladser. Resultatet kan derfor være "
            "afkortet. "
            f"Antal rækker: {len(baadpladser)}. "
            f"Maxrows: {valideret_maxrows}."
        )

    logger.info(
        "WebGIS returnerede %s bådpladser.",
        len(baadpladser),
    )

    return baadpladser

# ------------------------------------------------------------
# Validering
# ------------------------------------------------------------

def _valider_baadpladsraekker(
    baadpladser: Any,
) -> None:
    """Kontrollerer, at resultatet er en liste af dictionaries."""
    if not isinstance(
        baadpladser,
        list,
    ):
        raise TypeError(
            "hent_spatialanalyze_raekker() skal "
            "returnere en liste."
        )

    for index, baadplads in enumerate(
        baadpladser,
        start=1,
    ):
        if not isinstance(
            baadplads,
            dict,
        ):
            raise TypeError(
                "Alle bådpladsrækker skal være "
                "dictionaries. "
                f"Række {index} har typen "
                f"{type(baadplads).__name__}."
            )


def _valider_page(
    page: Page,
) -> None:
    """Kontrollerer, at Playwright-siden kan anvendes."""
    if page is None:
        raise ValueError(
            "page skal angives."
        )

    if page.is_closed():
        raise RuntimeError(
            "WebGIS-siden er lukket og kan ikke "
            "bruges til at hente bådpladser."
        )


def _valider_tekst(
    *,
    value: Any,
    field_name: str,
) -> str:
    """Returnerer en obligatorisk, trimmet tekstværdi."""
    if not isinstance(
        value,
        str,
    ):
        raise TypeError(
            f"{field_name} skal være tekst."
        )

    cleaned_value = value.strip()

    if not cleaned_value:
        raise ValueError(
            f"{field_name} må ikke være tom."
        )

    return cleaned_value


def _valider_positivt_heltal(
    *,
    value: Any,
    field_name: str,
) -> int:
    """Returnerer værdien som et positivt heltal."""
    if isinstance(
        value,
        bool,
    ):
        raise TypeError(
            f"{field_name} må ikke være boolsk."
        )

    try:
        integer_value = int(
            value
        )
    except (
        TypeError,
        ValueError,
    ) as exc:
        raise TypeError(
            f"{field_name} skal være et heltal."
        ) from exc

    if integer_value <= 0:
        raise ValueError(
            f"{field_name} skal være større end 0."
        )

    return integer_value

async def opret_webgis_rapport(
    page: Page,
    session: BrowserSession,
    wkt: str,
    *,
    tema: str = WEBGIS_STANDARD_TEMA,
    rapportmappe: str | Path = WEBGIS_RAPPORT_MAPPE,
    filnavn: str = WEBGIS_RAPPORT_FILNAVN,
    vaelg_jordstykker_byggesag: bool = True,
    timeout_ms: int = WEBGIS_TIMEOUT_MS,
    rapport_timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Path:
    """
    Udfører hele WebGIS-flowet og gemmer rapporten som PDF.

    BrowserSession modtages fra testen eller den kaldende proces.
    Funktionen opretter eller lukker ikke selve BrowserSession.

    Flow:
        1. Tager screenshot af WebGIS-startsiden.
        2. Vælger eventuelt temaet "Jordstykker - Byggesag".
        3. Åbner "Hvad gælder for" og indsætter WKT.
        4. Vælger og udfører konfliktsøgningen.
        5. Venter på status "Færdig".
        6. Åbner rapportdialogen.
        7. Klikker "Vis rapport".
        8. Gemmer PDF'en direkte fra browserens download-event.
        9. Tager screenshots efter hvert hovedtrin.

    Output:
        Path til den gemte PDF-fil.
    """

    _valider_page(
        page
    )

    if session is None:
        raise ValueError(
            "session skal angives."
        )

    valideret_wkt = _valider_wkt(
        wkt
    )

    valideret_tema = _valider_tema(
        tema
    )

    valideret_filnavn = _valider_pdf_filnavn(
        filnavn
    )

    destination = _opret_rapportsti(
        rapportmappe=rapportmappe,
        filnavn=valideret_filnavn,
    )

    rapport_side: Page | None = None

    try:
        logger.info(
            "Starter WebGIS-rapport for konfliktsøgning=%s.",
            valideret_tema,
        )

        await _tag_screenshot(
            session=session,
            page=page,
            navn="STEP_01_webgis_start",
        )

        if vaelg_jordstykker_byggesag:
            await vaelg_jordstykker_byggesag_i_webgis(
                page=page,
                timeout_ms=timeout_ms,
            )

            await _tag_screenshot(
                session=session,
                page=page,
                navn="STEP_02_jordstykker_byggesag_valgt",
            )

        await indsaet_wkt_i_webgis(
            page=page,
            wkt=valideret_wkt,
            timeout_ms=timeout_ms,
        )

        await _tag_screenshot(
            session=session,
            page=page,
            navn="STEP_03_wkt_indsaet",
        )

        await udfoer_konfliktsoegning(
            page=page,
            tema=valideret_tema,
            timeout_ms=rapport_timeout_ms,
        )

        await _tag_screenshot(
            session=session,
            page=page,
            navn="STEP_04_konfliktsoegning_faerdig",
        )

        rapport_side = await aabn_rapport(
            page=page,
            timeout_ms=rapport_timeout_ms,
        )

        await _tag_screenshot(
            session=session,
            page=rapport_side,
            navn="STEP_05_rapport_aabnet",
        )


        pdf_sti = await gem_rapport_pdf(
            webgis_side=page,
            rapport_side=rapport_side,
            destination=destination,
            timeout_ms=rapport_timeout_ms,
        )

        await _tag_screenshot(
            session=session,
            page=rapport_side,
            navn="STEP_06_pdf_rapport_gemt",
        )

        logger.info(
            "WebGIS-rapporten er gemt: %s",
            pdf_sti,
        )

        return pdf_sti

    except Exception:
        logger.exception(
            "WebGIS-rapportflowet fejlede."
        )

        screenshot_side = (
            rapport_side
            if (
                rapport_side is not None
                and not rapport_side.is_closed()
            )
            else page
        )

        await _tag_screenshot(
            session=session,
            page=screenshot_side,
            navn="FEJL_webgis_rapport",
        )

        raise

    finally:
        if rapport_side is not None:
            _RAPPORT_DOWNLOADS.pop(
                id(rapport_side),
                None,
            )

            _RAPPORT_PDF_BYTES.pop(
                id(rapport_side),
                None,
            )

        if (
            rapport_side is not None
            and rapport_side != page
            and not rapport_side.is_closed()
        ):
            await rapport_side.close()

async def vaelg_jordstykker_byggesag_i_webgis(
    page: Page,
    *,
    timeout_ms: int = WEBGIS_TIMEOUT_MS,
) -> None:
    """
    Søger efter temaet "Jordstykker - Byggesag".

    Efter søgeteksten er indsat, klikkes der igen i søgefeltet.
    Derefter klikkes der på søgeresultatgruppen "Temaer".

    Output:
        Funktionen returnerer ikke data.
        Temaresultatet bliver åbnet i WebGIS.
    """

    soegning = "Jordstykker - Byggesag"

    logger.info(
        "Søger efter temaet %s.",
        soegning,
    )

    soegefelt = page.locator(
        GLOBALT_SOEGEFELT
    ).first

    await _vent_paa_synlig(
        soegefelt,
        navn="globalt søgefelt",
        timeout_ms=timeout_ms,
    )

    await soegefelt.click()

    await _vent_mellem_selector_actions(
        page
    )

    await soegefelt.fill(
        soegning
    )

    await _vent_mellem_selector_actions(
        page
    )

    logger.info(
        "Søgeteksten er indsat. "
        "Klikker i søgefeltet igen."
    )

    await soegefelt.click()

    await _vent_mellem_selector_actions(
        page
    )

    tema_resultat = page.locator(
        TEMAER_RESULTAT
    ).first

    await _vent_paa_synlig(
        tema_resultat,
        navn='søgeresultatgruppen "Temaer"',
        timeout_ms=timeout_ms,
    )

    logger.info(
        'Søgeresultatgruppen "Temaer" er synlig. '
        "Klikker på resultatet."
    )

    await tema_resultat.click()

    await _vent_mellem_selector_actions(
        page
    )

    logger.info(
        'Søgeresultatgruppen "Temaer" er valgt '
        'efter søgning efter %s.',
        soegning,
    )

async def indsaet_wkt_i_webgis(
    page: Page,
    wkt: str,
    *,
    timeout_ms: int = WEBGIS_TIMEOUT_MS,
) -> None:
    """
    Åbner arbejdsgeometrien og indsætter WKT.

    Output:
        Funktionen returnerer ikke data. Geometrien indsættes i kortet.
    """

    valideret_wkt = _valider_wkt(wkt)

    logger.info(
        "Åbner WebGIS-arbejdsgeometrien."
    )

    arbejdsgeometri = page.locator(
        HVAD_GAELDER_FOR_MENU
    ).first

    await _vent_paa_synlig(
        arbejdsgeometri,
        navn="Hvad gælder for-menu",
        timeout_ms=timeout_ms,
    )

    # Klik på linket inde i li-elementet, hvis det findes.
    arbejdsgeometri_link = arbejdsgeometri.locator(
        'a[title="Hvad gælder for"]'
    )

    if await arbejdsgeometri_link.count() > 0:
        await arbejdsgeometri_link.click()

        await _vent_mellem_selector_actions(
            page
        )
    else:
        await arbejdsgeometri.click()

        await _vent_mellem_selector_actions(
            page
        )

    wkt_knap = page.locator(
        INDSAET_WKT_KNAP
    ).first

    await _vent_paa_synlig(
        wkt_knap,
        navn="Indsæt WKT i kortet",
        timeout_ms=timeout_ms,
    )

    await wkt_knap.click()

    await _vent_mellem_selector_actions(
        page
    )

    wkt_felt = page.locator(
        WKT_TEKSTFELT
    ).first

    await _vent_paa_synlig(
        wkt_felt,
        navn="WKT-tekstfelt",
        timeout_ms=timeout_ms,
    )

    await wkt_felt.fill(
        valideret_wkt
    )

    await _vent_mellem_selector_actions(
        page
    )

    if await wkt_felt.input_value() != valideret_wkt:
        raise RuntimeError(
            "WKT-tekstfeltet indeholder ikke den forventede WKT."
        )

    ok_knap = page.locator(
        WKT_OK_KNAP
    ).first

    await _vent_paa_aktiv(
        ok_knap,
        navn="OK-knappen til WKT",
        timeout_ms=timeout_ms,
    )

    await ok_knap.click()

    await _vent_mellem_selector_actions(
        page
    )

    hvad_gaelder_for = page.locator(
        HVAD_GAELDER_FOR_KNAP
    ).first

    await _vent_paa_aktiv(
        hvad_gaelder_for,
        navn="Hvad gælder for-knappen",
        timeout_ms=timeout_ms,
    )

    logger.info(
        "WKT er indsat i WebGIS."
    )

async def udfoer_konfliktsoegning(
    page: Page,
    *,
    tema: str,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> None:
    """
    Vælger en konfliktsøgning og venter på WebGIS-status "Færdig".
    """

    valideret_tema = _valider_tema(tema)

    hvad_gaelder_for = page.locator(
        HVAD_GAELDER_FOR_KNAP
    ).first

    await _vent_paa_aktiv(
        hvad_gaelder_for,
        navn="Hvad gælder for-knappen",
        timeout_ms=timeout_ms,
    )

    logger.info(
        "Åbner konfliktsøgningen."
    )

    await hvad_gaelder_for.click()

    await _vent_mellem_selector_actions(
        page
    )

    dropdown = page.locator(
        KONFLIKTSOEGNING_DROPDOWN
    ).first

    await _vent_paa_synlig(
        dropdown,
        navn="dropdown til konfliktsøgning",
        timeout_ms=timeout_ms,
    )

    await dropdown.select_option(
        value=valideret_tema
    )

    await _vent_mellem_selector_actions(
        page
    )

    valgt_tema = await dropdown.input_value()

    if valgt_tema != valideret_tema:
        raise RuntimeError(
            "WebGIS valgte ikke den forventede "
            f"konfliktsøgning. Forventede {valideret_tema!r}, "
            f"men fik {valgt_tema!r}."
        )

    udfoer_knap = page.locator(
        KONFLIKTSOEGNING_UDFOER_KNAP
    ).first

    await _vent_paa_aktiv(
        udfoer_knap,
        navn="Udfør-knappen",
        timeout_ms=timeout_ms,
    )

    logger.info(
        "Starter konfliktsøgning=%s.",
        valideret_tema,
    )

    await udfoer_knap.click()

    await _vent_mellem_selector_actions(
        page
    )

    await _vent_paa_konfliktsoegning_faerdig(
        page=page,
        timeout_ms=timeout_ms,
    )

    logger.info(
        "Konfliktsøgningen er færdig."
    )


async def _vent_paa_konfliktsoegning_faerdig(
    page: Page,
    *,
    timeout_ms: int,
) -> None:
    """
    Venter på status "Færdig" og kontrollerer, at loading-logoet er væk.
    """

    status = page.locator(
        KONFLIKTSOEGNING_STATUS
    ).first

    try:
        await status.wait_for(
            state="visible",
            timeout=timeout_ms,
        )

        await status.filter(
            has_text=re.compile(
                r"^\s*Færdig\s*$",
                re.IGNORECASE,
            )
        ).wait_for(
            state="visible",
            timeout=timeout_ms,
        )
    except PlaywrightTimeoutError as exc:
        status_tekst = ""

        if await status.count() > 0:
            status_tekst = (
                await status.text_content()
                or ""
            ).strip()

        raise RuntimeError(
            "WebGIS-konfliktsøgningen blev ikke færdig. "
            f"Seneste synlige status: {status_tekst!r}."
        ) from exc

    loading = page.locator(
        KONFLIKTSOEGNING_LOADING
    ).first

    if await loading.count() > 0:
        try:
            await loading.wait_for(
                state="hidden",
                timeout=timeout_ms,
            )
        except PlaywrightTimeoutError as exc:
            raise RuntimeError(
                "WebGIS viste status 'Færdig', men "
                "loading-logoet forsvandt ikke."
            ) from exc

async def aabn_rapport(
    page: Page,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Page:
    """
    Bestiller WebGIS-rapporten og venter på et PDF-resultat.

    Funktionen bevarer det eksisterende download-flow og tilføjer
    to målrettede sikkerhedsnet:

    1. Et PDF-response accepteres, hvis Chromium ikke udløser et
       normalt download-event.
    2. Rapportbestillingen forsøges én ekstra gang, hvis første klik
       ikke giver et download eller PDF-response.

    Den samlede ventetid overstiger ikke timeout_ms.

    Output:
        Returnerer den side, som rapportresultatet knyttes til.
        Resultatet gemmes midlertidigt i _RAPPORT_DOWNLOADS eller
        _RAPPORT_PDF_BYTES og anvendes af gem_rapport_pdf().
    """
    _valider_page(page)

    if (
        isinstance(timeout_ms, bool)
        or not isinstance(timeout_ms, int)
    ):
        raise TypeError(
            "timeout_ms skal være et heltal."
        )

    if timeout_ms <= 0:
        raise ValueError(
            "timeout_ms skal være større end 0."
        )

    rapport_knap = page.locator(
        RAPPORT_KNAP
    ).filter(
        has_text=re.compile(
            r"^\s*Rapport\s*$",
            re.IGNORECASE,
        )
    ).first

    await _vent_paa_synlig(
        rapport_knap,
        navn="Rapport-knappen",
        timeout_ms=timeout_ms,
    )

    logger.info(
        "Åbner WebGIS-rapportdialogen."
    )

    await rapport_knap.click()

    await _vent_mellem_selector_actions(
        page
    )

    vis_rapport_knap = page.locator(
        VIS_RAPPORT_KNAP
    ).filter(
        has_text=re.compile(
            r"^\s*Vis rapport\s*$",
            re.IGNORECASE,
        )
    ).first

    await _vent_paa_aktiv(
        vis_rapport_knap,
        navn="Vis rapport-knappen",
        timeout_ms=timeout_ms,
    )

    loading = page.locator(
        RAPPORT_LOADING
    ).first

    event_loop = asyncio.get_running_loop()

    download_future: asyncio.Future[Download] = (
        event_loop.create_future()
    )

    pdf_response_future: asyncio.Future[bytes] = (
        event_loop.create_future()
    )

    nye_sider: list[Page] = []
    sider_med_download_listener: list[Page] = [page]
    response_tasks: set[asyncio.Task[None]] = set()
    netvaerksfejl: list[str] = []
    mulige_pdf_svar: list[str] = []

    def ved_download(
        download: Download,
    ) -> None:
        if not download_future.done():
            download_future.set_result(download)

    async def behandl_response(response) -> None:
        try:
            headers = await response.all_headers()
            content_type = str(
                headers.get("content-type", "")
            ).casefold()
            content_disposition = str(
                headers.get("content-disposition", "")
            ).casefold()
            response_url = str(response.url or "")

            er_pdf_response = (
                "application/pdf" in content_type
                or (
                    "filename=" in content_disposition
                    and ".pdf" in content_disposition
                )
                or re.search(
                    r"\.pdf(?:$|[?#])",
                    response_url,
                    re.IGNORECASE,
                ) is not None
            )

            if not er_pdf_response:
                return

            mulige_pdf_svar.append(
                _sikker_url(response_url)
            )

            pdf_bytes = await response.body()

            if not pdf_bytes.startswith(b"%PDF-"):
                logger.warning(
                    "WebGIS returnerede et muligt PDF-response "
                    "uden gyldig PDF-signatur. URL=%s, "
                    "Content-Type=%s.",
                    _sikker_url(response_url),
                    content_type,
                )
                return

            if not pdf_response_future.done():
                pdf_response_future.set_result(pdf_bytes)

        except Exception:
            logger.debug(
                "Et WebGIS-response kunne ikke undersøges.",
                exc_info=True,
            )

    def ved_response(response) -> None:
        task = event_loop.create_task(
            behandl_response(response)
        )
        response_tasks.add(task)
        task.add_done_callback(
            response_tasks.discard
        )

    def ved_request_failed(request) -> None:
        try:
            netvaerksfejl.append(
                f"{request.method} "
                f"{_sikker_url(request.url)} "
                f"failure={request.failure!r}"
            )
        except Exception:
            netvaerksfejl.append(
                "Et WebGIS-request fejlede uden læsbare detaljer."
            )

    def ved_ny_side(ny_side: Page) -> None:
        nye_sider.append(ny_side)
        ny_side.on("download", ved_download)
        sider_med_download_listener.append(ny_side)

        logger.info(
            "WebGIS oprettede en rapportfane. "
            "Aktuel URL: %s",
            _sikker_url(ny_side.url),
        )

    page.on("download", ved_download)
    page.context.on("page", ved_ny_side)
    page.context.on("response", ved_response)
    page.context.on("requestfailed", ved_request_failed)

    async def vent_paa_rapportresultat(
        ventetid_ms: int,
    ) -> tuple[str, Download | bytes]:
        faerdige, _ = await asyncio.wait(
            {
                download_future,
                pdf_response_future,
            },
            timeout=ventetid_ms / 1_000,
            return_when=asyncio.FIRST_COMPLETED,
        )

        if download_future in faerdige:
            return (
                "download",
                download_future.result(),
            )

        if pdf_response_future in faerdige:
            return (
                "pdf_bytes",
                pdf_response_future.result(),
            )

        raise TimeoutError(
            "Der kom hverken download-event eller PDF-response."
        )

    async def hent_synlig_tekst() -> str:
        try:
            return " ".join(
                (
                    await page.locator("body").inner_text()
                ).split()
            )
        except Exception:
            return ""

    try:
        foerste_forsoeg_ms = max(
            1,
            int(timeout_ms * 0.60),
        )
        andet_forsoeg_ms = max(
            1,
            timeout_ms - foerste_forsoeg_ms,
        )

        logger.info(
            "Bestiller rapporten og venter på download-event "
            "eller PDF-response. Første forsøg=%s ms, "
            "genforsøg=%s ms.",
            foerste_forsoeg_ms,
            andet_forsoeg_ms,
        )

        await vis_rapport_knap.click()

        try:
            resultat_type, resultat = (
                await vent_paa_rapportresultat(
                    foerste_forsoeg_ms
                )
            )

        except TimeoutError:
            logger.warning(
                "Første rapportbestilling gav hverken download, "
                "PDF-response eller rapportfane. Udfører ét "
                "kontrolleret genforsøg."
            )

            if not await vis_rapport_knap.is_visible():
                await _vent_paa_synlig(
                    rapport_knap,
                    navn="Rapport-knappen ved genforsøg",
                    timeout_ms=andet_forsoeg_ms,
                )

                await rapport_knap.click()

                await _vent_mellem_selector_actions(
                    page
                )

                vis_rapport_knap = page.locator(
                    VIS_RAPPORT_KNAP
                ).filter(
                    has_text=re.compile(
                        r"^\s*Vis rapport\s*$",
                        re.IGNORECASE,
                    )
                ).first

            overlay_arbejder = (
                await loading.count() > 0
                and await loading.is_visible()
            )

            if overlay_arbejder:
                logger.warning(
                    "WebGIS viser stadig 'Arbejder...'. "
                    "Den oprindelige rapportbestilling kører "
                    "derfor fortsat. Klikker ikke Vis rapport "
                    "igen, men venter videre på download eller "
                    "PDF-response."
                )

            else:
                await _vent_paa_aktiv(
                    vis_rapport_knap,
                    navn="Vis rapport-knappen ved genforsøg",
                    timeout_ms=andet_forsoeg_ms,
                )

                logger.info(
                    "WebGIS arbejder ikke længere. "
                    "Klikker Vis rapport igen."
                )

                await vis_rapport_knap.click(
                    timeout=andet_forsoeg_ms,
                )

            try:
                resultat_type, resultat = (
                    await vent_paa_rapportresultat(
                        andet_forsoeg_ms
                    )
                )

            except TimeoutError as error:
                popup_information = [
                    {
                        "url": _sikker_url(popup.url),
                        "lukket": popup.is_closed(),
                    }
                    for popup in nye_sider
                ]

                synlig_tekst = await hent_synlig_tekst()

                raise RuntimeError(
                    "WebGIS leverede ikke en PDF efter to "
                    "rapportbestillinger. "
                    f"Samlet timeout={timeout_ms} ms. "
                    f"Registrerede rapportfaner="
                    f"{popup_information}. "
                    f"Mulige PDF-responses="
                    f"{mulige_pdf_svar[-10:]}. "
                    f"Seneste netværksfejl="
                    f"{netvaerksfejl[-10:]}. "
                    f"Synlig WebGIS-tekst="
                    f"{synlig_tekst[:2_000]!r}."
                ) from error

        rapport_side = page

        for mulig_rapport_side in reversed(nye_sider):
            if not mulig_rapport_side.is_closed():
                rapport_side = mulig_rapport_side
                break

        if resultat_type == "download":
            download = resultat

            if not isinstance(download, Download):
                raise RuntimeError(
                    "Det registrerede downloadresultat "
                    "havde en uventet type."
                )

            download_fejl = await download.failure()

            if download_fejl:
                raise RuntimeError(
                    "WebGIS-downloadet fejlede: "
                    f"{download_fejl}"
                )

            _RAPPORT_DOWNLOADS[
                id(rapport_side)
            ] = download

            logger.info(
                "Download-event modtaget. Foreslået filnavn: %s. "
                "Rapportfane oprettet: %s.",
                download.suggested_filename,
                rapport_side is not page,
            )

        elif resultat_type == "pdf_bytes":
            pdf_bytes = resultat

            if not isinstance(pdf_bytes, bytes):
                raise RuntimeError(
                    "Det registrerede PDF-resultat var ikke bytes."
                )

            if not pdf_bytes.startswith(b"%PDF-"):
                raise RuntimeError(
                    "WebGIS-response mangler en gyldig "
                    "PDF-signatur."
                )

            _RAPPORT_PDF_BYTES[
                id(rapport_side)
            ] = pdf_bytes

            logger.info(
                "PDF-response modtaget direkte fra WebGIS: "
                "%s bytes. Rapportfane oprettet: %s.",
                len(pdf_bytes),
                rapport_side is not page,
            )

        else:
            raise RuntimeError(
                "WebGIS returnerede en ukendt rapporttype: "
                f"{resultat_type!r}."
            )

        if await loading.count() > 0:
            try:
                await loading.wait_for(
                    state="hidden",
                    timeout=min(timeout_ms, 60_000),
                )
            except PlaywrightTimeoutError:
                logger.debug(
                    "Rapport-loading forsvandt ikke, men "
                    "PDF-resultatet er allerede modtaget."
                )

        return rapport_side

    finally:
        page.context.remove_listener(
            "page",
            ved_ny_side,
        )
        page.context.remove_listener(
            "response",
            ved_response,
        )
        page.context.remove_listener(
            "requestfailed",
            ved_request_failed,
        )

        for listener_side in sider_med_download_listener:
            try:
                listener_side.remove_listener(
                    "download",
                    ved_download,
                )
            except Exception:
                logger.debug(
                    "Download-listeneren kunne ikke fjernes "
                    "fra en WebGIS-side.",
                    exc_info=True,
                )

        if response_tasks:
            await asyncio.gather(
                *response_tasks,
                return_exceptions=True,
            )

async def gem_rapport_pdf(
    webgis_side: Page,
    rapport_side: Page,
    destination: str | Path,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Path:
    """
    Gemmer WebGIS-rapporten som PDF.

    Rapporten kan komme fra et normalt Playwright-download eller fra
    et direkte PDF-response. Der foretages ikke et nyt HTTP-kald.

    Output:
        Path til den gemte og validerede PDF-fil.
    """
    _valider_page(webgis_side)
    _valider_page(rapport_side)

    if (
        isinstance(timeout_ms, bool)
        or not isinstance(timeout_ms, int)
    ):
        raise TypeError(
            "timeout_ms skal være et heltal."
        )

    if timeout_ms <= 0:
        raise ValueError(
            "timeout_ms skal være større end 0."
        )

    download = _RAPPORT_DOWNLOADS.pop(
        id(rapport_side),
        None,
    )

    pdf_bytes = _RAPPORT_PDF_BYTES.pop(
        id(rapport_side),
        None,
    )

    if download is None and pdf_bytes is None:
        raise RuntimeError(
            "Der blev hverken registreret et download eller "
            "et PDF-response til den aktuelle WebGIS-rapport."
        )

    destination_path = Path(destination)

    destination_path.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    destination_path.unlink(
        missing_ok=True
    )

    if download is not None:
        logger.info(
            "Gemmer PDF direkte fra browserens download-event."
        )

        await download.save_as(
            str(destination_path)
        )

    else:
        if not isinstance(pdf_bytes, bytes):
            raise RuntimeError(
                "WebGIS PDF-response havde en uventet datatype."
            )

        logger.info(
            "Gemmer PDF direkte fra WebGIS-response."
        )

        destination_path.write_bytes(pdf_bytes)

    if not destination_path.is_file():
        raise RuntimeError(
            "WebGIS-rapporten blev ikke gemt."
        )

    gemt_filstoerrelse = destination_path.stat().st_size

    if gemt_filstoerrelse == 0:
        destination_path.unlink(missing_ok=True)
        raise RuntimeError(
            "Den gemte WebGIS-rapport er tom."
        )

    with destination_path.open("rb") as pdf_fil:
        pdf_signatur = pdf_fil.read(5)

    if pdf_signatur != b"%PDF-":
        destination_path.unlink(missing_ok=True)
        raise RuntimeError(
            "Den gemte fil mangler en gyldig PDF-signatur."
        )

    logger.info(
        "WebGIS-PDF gemt og valideret: %s bytes.",
        gemt_filstoerrelse,
    )

    return destination_path

async def _vent_paa_pdf_url(
    rapport_side: Page,
    *,
    timeout_ms: int,
) -> None:
    """

            re.compile(
                r"\.pdf(?:$|[?#])",
                re.IGNORECASE,
            ),
            timeout=timeout_ms,
        )
    except PlaywrightTimeoutError as exc:
        raise RuntimeError(
            "Rapportfanen blev åbnet, men navigerede ikke "
            "til en PDF-adresse. "
            f"Aktuel URL: {_sikker_url(rapport_side.url)}"
        ) from exc

async def _vent_mellem_selector_actions(
    page: Page,
    *,
    ventetid_ms: int = WEBGIS_SELECTOR_ACTION_WAIT_MS,
) -> None:
    """
    Indsætter en fast pause mellem Playwright-selectorhandlinger.

    Funktionen bruges efter click(), fill(), press() og
    select_option(), så WebGIS får tid til at opdatere
    brugerfladen.

    Output:
        Funktionen returnerer ikke data.
    """

    _valider_page(
        page
    )

    if isinstance(
        ventetid_ms,
        bool,
    ) or not isinstance(
        ventetid_ms,
        int,
    ):
        raise TypeError(
            "ventetid_ms skal være et heltal."
        )

    if ventetid_ms < 0:
        raise ValueError(
            "ventetid_ms må ikke være negativ."
        )

    if ventetid_ms == 0:
        return

    await page.wait_for_timeout(
        ventetid_ms
    )

async def _vent_paa_synlig(
    locator: Locator,
    *,
    navn: str,
    timeout_ms: int,
) -> None:
    """
    Venter på, at et element bliver synligt.
    """

    try:
        await locator.wait_for(
            state="visible",
            timeout=timeout_ms,
        )
    except PlaywrightTimeoutError as exc:
        raise RuntimeError(
            f"WebGIS-elementet {navn!r} blev ikke synligt."
        ) from exc


async def _vent_paa_aktiv(
    locator: Locator,
    *,
    navn: str,
    timeout_ms: int,
) -> None:
    """
    Venter på, at et element er synligt og aktiveret.
    """

    await _vent_paa_synlig(
        locator,
        navn=navn,
        timeout_ms=timeout_ms,
    )

    try:
        await locator.wait_for(
            state="attached",
            timeout=timeout_ms,
        )

        await locator.evaluate(
            """
            (element, timeoutMs) => new Promise(
                (resolve, reject) => {
                    const startedAt = Date.now();

                    const check = () => {
                        const disabled = (
                            element.disabled === true
                            || element.getAttribute(
                                "aria-disabled"
                            ) === "true"
                        );

                        if (!disabled) {
                            resolve(true);
                            return;
                        }

                        if (
                            Date.now() - startedAt
                            >= timeoutMs
                        ) {
                            reject(
                                new Error(
                                    "Elementet forblev disabled."
                                )
                            );
                            return;
                        }

                        setTimeout(check, 100);
                    };

                    check();
                }
            )
            """,
            timeout_ms,
        )
    except Exception as exc:
        raise RuntimeError(
            f"WebGIS-elementet {navn!r} blev ikke aktivt."
        ) from exc

def _valider_page(
    page: Page,
) -> None:
    """
    Kontrollerer, at WebGIS-siden kan bruges.
    """

    if page is None:
        raise ValueError(
            "page skal angives."
        )

    if page.is_closed():
        raise RuntimeError(
            "WebGIS-siden er lukket."
        )


def _valider_wkt(
    wkt: str,
) -> str:
    """
    Kontrollerer en grundlæggende WKT-værdi.

    Output:
        Trimmet WKT.
    """

    if not isinstance(wkt, str):
        raise TypeError(
            "wkt skal være tekst."
        )

    valideret_wkt = wkt.strip()

    if not valideret_wkt:
        raise ValueError(
            "wkt må ikke være tom."
        )

    tilladte_geometrityper = (
        "POINT",
        "LINESTRING",
        "POLYGON",
        "MULTIPOINT",
        "MULTILINESTRING",
        "MULTIPOLYGON",
        "GEOMETRYCOLLECTION",
    )

    wkt_upper = valideret_wkt.upper()

    if not wkt_upper.startswith(
        tilladte_geometrityper
    ):
        raise ValueError(
            "wkt starter ikke med en understøttet "
            "WKT-geometritype."
        )

    if (
        valideret_wkt.count("(")
        != valideret_wkt.count(")")
    ):
        raise ValueError(
            "wkt har ikke samme antal åbne og lukkede parenteser."
        )

    return valideret_wkt


def _valider_tema(
    tema: str,
) -> str:
    """
    Kontrollerer dropdown-værdien.

    Output:
        Trimmet temaværdi.
    """

    if not isinstance(tema, str):
        raise TypeError(
            "tema skal være tekst."
        )

    valideret_tema = tema.strip()

    if valideret_tema not in WEBGIS_TILLADTE_TEMAER:
        raise ValueError(
            f"Ukendt WebGIS-tema: {tema!r}. "
            f"Tilladte værdier: {WEBGIS_TILLADTE_TEMAER}."
        )

    return valideret_tema


def _valider_pdf_filnavn(
    filnavn: str,
) -> str:
    """
    Kontrollerer rapportens filnavn.

    Output:
        Et sikkert filnavn, som ender på .pdf.
    """

    if not isinstance(filnavn, str):
        raise TypeError(
            "filnavn skal være tekst."
        )

    valideret_filnavn = filnavn.strip()

    if not valideret_filnavn:
        raise ValueError(
            "filnavn må ikke være tomt."
        )

    if Path(valideret_filnavn).name != valideret_filnavn:
        raise ValueError(
            "filnavn må ikke indeholde en mappesti."
        )

    if not valideret_filnavn.casefold().endswith(
        ".pdf"
    ):
        valideret_filnavn += ".pdf"

    return valideret_filnavn


def _opret_rapportsti(
    *,
    rapportmappe: str | Path,
    filnavn: str,
) -> Path:
    """
    Opretter rapportmappen.

    Output:
        Den komplette sti til PDF-filen.
    """

    rapportmappe_path = Path(
        rapportmappe
    )

    rapportmappe_path.mkdir(
        parents=True,
        exist_ok=True,
    )

    return (
        rapportmappe_path
        / filnavn
    )


def _sikker_url(
    url: str,
) -> str:
    """
    Returnerer URL uden query-parametre eller credentials.
    """

    parsed_url = urlparse(
        str(url or "")
    )

    return parsed_url._replace(
        query="",
        fragment="",
        params="",
    ).geturl()

from __future__ import annotations

import logging
from typing import Any

    "https://webgis.intern.udcit.dk/"
    "spatialmap?page=spatialanalyze"

async def spatialanalyze(
    page: Page,
    session_id: str,
    datasource: str,
    min_x_wc: float,
    min_y_wc: float,
    max_x_wc: float,
    max_y_wc: float,
    maxrows: int,
    *,
    command: str,
    spatialexpression: str,
    jsontypes: str,
    outputformat: str,
    error_content_type: str,
    profile: str = "maxi",
    timeout_ms: int = DEFAULT_TIMEOUT_MS,
) -> dict[str, Any]:
    """Udfører WebGIS spatialanalyze i den åbne browser.

    Kaldet sendes som application/x-www-form-urlencoded gennem
    browserens egen fetch-funktion. Derfor genbruges browserens
    eksisterende HTTP-login og JSESSIONID.

    Parametre:
        page:
            Den allerede åbnede og autentificerede WebGIS-side.

        session_id:
            Session-Id opsnappet fra WebGIS' egne requests.

        datasource:
            WebGIS-datasourcen, eksempelvis
            "ds-byggesag_cases_matrikel".

        min_x_wc, min_y_wc, max_x_wc, max_y_wc:
            Koordinaterne til analyseområdet.

        maxrows:
            Maksimalt antal resultatrækker.

        command:
            Den præcise command-værdi fra Request 21.

        spatialexpression:
            Den præcise spatialexpression-værdi fra Request 21.

        jsontypes:
            Den præcise jsontypes-værdi fra Request 21.

        outputformat:
            Den præcise outputformat-værdi fra Request 21.

        error_content_type:
            Den præcise værdi for jdaf.error.contenttype
            fra Request 21.

        profile:
            WebGIS-profil. Standarden er "maxi".

        timeout_ms:
            Maksimal ventetid for browserkaldet.

    Output:
        WebGIS-svaret som en dictionary.

    Sikkerhed:
        Session-Id, cookies og formularindhold logges ikke.
    """
    _valider_page(page)

    session_id = _valider_tekst(
        value=session_id,
        field_name="session_id",
    )

    datasource = _valider_tekst(
        value=datasource,
        field_name="datasource",
    )

    command = _valider_tekst(
        value=command,
        field_name="command",
    )

    spatialexpression = _valider_tekst(
        value=spatialexpression,
        field_name="spatialexpression",
    )

    jsontypes = _valider_tekst(
        value=jsontypes,
        field_name="jsontypes",
    )

    outputformat = _valider_tekst(
        value=outputformat,
        field_name="outputformat",
    )

    error_content_type = _valider_tekst(
        value=error_content_type,
        field_name="error_content_type",
    )

    profile = _valider_tekst(
        value=profile,
        field_name="profile",
    )

    maxrows = _valider_positivt_heltal(
        value=maxrows,
        field_name="maxrows",
    )

    timeout_ms = _valider_positivt_heltal(
        value=timeout_ms,
        field_name="timeout_ms",
    )

    koordinater = _valider_koordinater(
        min_x_wc=min_x_wc,
        min_y_wc=min_y_wc,
        max_x_wc=max_x_wc,
        max_y_wc=max_y_wc,
    )

    formular_data = {
        "page": "spatialanalyze",
        "profile": profile,
        "command": command,
        "datasource": datasource,
        "spatialexpression": spatialexpression,
        "min_x_wc": str(koordinater["min_x_wc"]),
        "min_y_wc": str(koordinater["min_y_wc"]),
        "max_x_wc": str(koordinater["max_x_wc"]),
        "max_y_wc": str(koordinater["max_y_wc"]),
        "maxrows": str(maxrows),
        "outputformat": outputformat,
        "jsontypes": jsontypes,
        "jdaf.error.contenttype": error_content_type,
    }

    logger.info(
        "Starter WebGIS spatialanalyze for datasource=%s, "
        "maxrows=%s.",
        datasource,
        maxrows,
    )

    browser_resultat = await page.evaluate(
        """
        async ({
            url,
            sessionId,
            formValues,
            timeoutMs
        }) => {
            const controller = new AbortController();

            const timeoutHandle = setTimeout(
                () => controller.abort(),
                timeoutMs
            );

            try {
                const formBody = new URLSearchParams();

                for (
                    const [fieldName, fieldValue]
                    of Object.entries(formValues)
                ) {
                    formBody.append(
                        fieldName,
                        String(fieldValue)
                    );
                }

                const response = await fetch(
                    url,
                    {
                        method: "POST",
                        credentials: "include",
                        headers: {
                            "Accept": (
                                "application/json, "
                                + "text/javascript, "
                                + "*/*; q=0.01"
                            ),
                            "Content-Type": (
                                "application/"
                                + "x-www-form-urlencoded; "
                                + "charset=UTF-8"
                            ),
                            "Session-Id": sessionId,
                            "X-Requested-With": (
                                "XMLHttpRequest"
                            )
                        },
                        body: formBody.toString(),
                        signal: controller.signal
                    }
                );

                const responseText =
                    await response.text();

                return {
                    ok: response.ok,
                    status: response.status,
                    statusText: response.statusText,
                    contentType: (
                        response.headers.get(
                            "content-type"
                        ) || ""
                    ),
                    responseText: responseText,
                    finalUrl: response.url
                };
            } catch (error) {
                return {
                    ok: false,
                    status: 0,
                    statusText: "",
                    contentType: "",
                    responseText: "",
                    finalUrl: "",
                    browserError: (
                        error instanceof Error
                            ? (
                                error.name
                                + ": "
                                + error.message
                            )
                            : String(error)
                    )
                };
            } finally {
                clearTimeout(timeoutHandle);
            }
        }
        """,
        {
            "url": SPATIALANALYZE_URL,
            "sessionId": session_id,
            "formValues": formular_data,
            "timeoutMs": timeout_ms,
        },
    )

    fetch_resultat = _valider_fetch_resultat(
        browser_resultat
    )

    status = fetch_resultat["status"]

    browser_error = str(
        fetch_resultat.get("browserError")
        or ""
    ).strip()

    if browser_error:
        raise RuntimeError(
            "WebGIS spatialanalyze kunne ikke udføres "
            "i browseren. "
            f"Browserfejl: {browser_error}"
        )

    if not fetch_resultat["ok"]:
        response_preview = _lav_sikkert_response_preview(
            fetch_resultat.get("responseText")
        )

        raise RuntimeError(
            "WebGIS spatialanalyze fejlede med "
            f"HTTP-status {status}. "
            f"Statusbesked: "
            f"{fetch_resultat.get('statusText')!s}. "
            f"Svareksempel: {response_preview}"
        )

    response_text = str(
        fetch_resultat.get("responseText")
        or ""
    ).strip()

    if not response_text:
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede "
            "et tomt svar."
        )

    resultat = await _parse_json_i_browser(
        page=page,
        response_text=response_text,
    )

    if not isinstance(resultat, dict):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede JSON, "
            "men resultatet var ikke en dictionary. "
            f"Modtog: {type(resultat).__name__}."
        )

    logger.info(
        "WebGIS spatialanalyze blev gennemført med "
        "HTTP-status %s.",
        status,
    )

    return resultat


# ------------------------------------------------------------
# JSON
# ------------------------------------------------------------


async def _parse_json_i_browser(
    page: Page,
    response_text: str,
) -> Any:
    """Parser WebGIS-svaret som JSON.

    Output:
        Den parsede JSON-værdi.
    """
    parse_resultat = await page.evaluate(
        """
        (responseText) => {
            try {
                return {
                    parsed: true,
                    value: JSON.parse(responseText)
                };
            } catch (error) {
                return {
                    parsed: false,
                    error: (
                        error instanceof Error
                            ? error.message
                            : String(error)
                    )
                };
            }
        }
        """,
        response_text,
    )

    if not isinstance(parse_resultat, dict):
        raise RuntimeError(
            "WebGIS-svaret kunne ikke kontrolleres "
            "som JSON."
        )

    if not parse_resultat.get("parsed"):
        response_preview = (
            _lav_sikkert_response_preview(
                response_text
            )
        )

        raise RuntimeError(
            "WebGIS spatialanalyze returnerede ikke "
            "gyldig JSON. "
            f"JSON-fejl: "
            f"{parse_resultat.get('error')!s}. "
            f"Svareksempel: {response_preview}"
        )

    return parse_resultat.get("value")


# ------------------------------------------------------------
# Validering af fetch-resultat
# ------------------------------------------------------------


def _valider_fetch_resultat(
    value: Any,
) -> dict[str, Any]:
    """Validerer resultatet fra browserens fetch-kald."""
    if not isinstance(value, dict):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede et "
            "ugyldigt browserresultat."
        )

    status = value.get("status")

    if isinstance(status, bool):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede en "
            "ugyldig HTTP-status."
        )

    if not isinstance(status, int):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede ikke "
            "en numerisk HTTP-status."
        )

    if not isinstance(value.get("ok"), bool):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede ikke "
            "en gyldig ok-status."
        )

    return value

def _valider_page(
    page: Page,
) -> None:
    """Kontrollerer, at browsersiden kan bruges."""
    if page is None:
        raise ValueError(
            "page skal angives."
        )

    if page.is_closed():
        raise RuntimeError(
            "WebGIS-siden er lukket og kan ikke "
            "bruges til spatialanalyze."
        )


def _valider_tekst(
    *,
    value: Any,
    field_name: str,
) -> str:
    """Returnerer en obligatorisk, trimmet tekstværdi."""
    if not isinstance(value, str):
        raise TypeError(
            f"{field_name} skal være tekst."
        )

    cleaned_value = value.strip()

    if not cleaned_value:
        raise ValueError(
            f"{field_name} må ikke være tom."
        )

    return cleaned_value


def _valider_positivt_heltal(
    *,
    value: Any,
    field_name: str,
) -> int:
    """Returnerer værdien som et positivt heltal."""
    if isinstance(value, bool):
        raise TypeError(
            f"{field_name} må ikke være boolsk."
        )

    try:
        integer_value = int(value)
    except (TypeError, ValueError) as exc:
        raise TypeError(
            f"{field_name} skal være et heltal."
        ) from exc

    if integer_value <= 0:
        raise ValueError(
            f"{field_name} skal være større end 0."
        )

    return integer_value


def _valider_koordinat(
    *,
    value: Any,
    field_name: str,
) -> float:
    """Returnerer en koordinat som float."""
    if isinstance(value, bool):
        raise TypeError(
            f"{field_name} må ikke være boolsk."
        )

    try:
        return float(value)
    except (TypeError, ValueError) as exc:
        raise TypeError(
            f"{field_name} skal være et tal."
        ) from exc


def _valider_koordinater(
    *,
    min_x_wc: Any,
    min_y_wc: Any,
    max_x_wc: Any,
    max_y_wc: Any,
) -> dict[str, float]:
    """Validerer spatialanalysens koordinater."""
    koordinater = {
        "min_x_wc": _valider_koordinat(
            value=min_x_wc,
            field_name="min_x_wc",
        ),
        "min_y_wc": _valider_koordinat(
            value=min_y_wc,
            field_name="min_y_wc",
        ),
        "max_x_wc": _valider_koordinat(
            value=max_x_wc,
            field_name="max_x_wc",
        ),
        "max_y_wc": _valider_koordinat(
            value=max_y_wc,
            field_name="max_y_wc",
        ),
    }

    if (
        koordinater["min_x_wc"]
        >= koordinater["max_x_wc"]
    ):
        raise ValueError(
            "min_x_wc skal være mindre end max_x_wc."
        )

    if (
        koordinater["min_y_wc"]
        >= koordinater["max_y_wc"]
    ):
        raise ValueError(
            "min_y_wc skal være mindre end max_y_wc."
        )

    return koordinater

def _lav_sikkert_response_preview(
    value: Any,
    maximum_length: int = 500,
) -> str:
    """Laver et begrænset preview af et fejlsvar.

    Previewet bruges kun til fejlsøgning. Session-Id og cookies
    ligger ikke i response-bodyen og bliver ikke tilføjet her.
    """
    text = " ".join(
        str(value or "").split()
    )

    if not text:
        return "<tomt svar>"

    if len(text) <= maximum_length:
        return text

    return (
        text[:maximum_length]
        + "..."
    )

async def hent_spatialanalyze_raekker(
    page: Page,
    session_id: str,
    *,
    maxrows: int,
    jsontypes: str,
    outputformat: str,
    error_content_type: str,
    timeout_ms: int = DEFAULT_TIMEOUT_MS,
) -> list[dict[str, Any]]:
    """Henter rækkerne fra den seneste spatialanalyse.

    Funktionen skal kaldes umiddelbart efter spatialanalyze(),
    mens den samme WebGIS-side og Session-Id stadig er aktive.

    Output:
        Liste med de konkrete WebGIS-rækker som dictionaries.

    Sikkerhed:
        Session-Id og formularværdier bliver ikke skrevet i loggen.
    """
    _valider_page(page)

    session_id = _valider_tekst(
        value=session_id,
        field_name="session_id",
    )

    jsontypes = _valider_tekst(
        value=jsontypes,
        field_name="jsontypes",
    )

    outputformat = _valider_tekst(
        value=outputformat,
        field_name="outputformat",
    )

    error_content_type = _valider_tekst(
        value=error_content_type,
        field_name="error_content_type",
    )

    maxrows = _valider_positivt_heltal(
        value=maxrows,
        field_name="maxrows",
    )

    timeout_ms = _valider_positivt_heltal(
        value=timeout_ms,
        field_name="timeout_ms",
    )

    formular_data = {
        "page": "spatialanalyze-get-stored-data",
        "maxrows": str(maxrows),
        "outputformat": outputformat,
        "jsontypes": jsontypes,
        "jdaf.error.contenttype": error_content_type,
    }

    logger.info(
        "Henter op til %s rækker fra den seneste "
        "WebGIS-spatialanalyse.",
        maxrows,
    )

    browser_resultat = await page.evaluate(
        """
        async ({
            url,
            sessionId,
            formValues,
            timeoutMs
        }) => {
            const controller = new AbortController();

            const timeoutHandle = setTimeout(
                () => controller.abort(),
                timeoutMs
            );

            try {
                const formBody = new URLSearchParams();

                for (
                    const [fieldName, fieldValue]
                    of Object.entries(formValues)
                ) {
                    formBody.append(
                        fieldName,
                        String(fieldValue)
                    );
                }

                const response = await fetch(
                    url,
                    {
                        method: "POST",
                        credentials: "include",
                        headers: {
                            "Accept": (
                                "application/json, "
                                + "text/javascript, "
                                + "*/*; q=0.01"
                            ),
                            "Content-Type": (
                                "application/"
                                + "x-www-form-urlencoded; "
                                + "charset=UTF-8"
                            ),
                            "Session-Id": sessionId,
                            "X-Requested-With": "XMLHttpRequest"
                        },
                        body: formBody.toString(),
                        signal: controller.signal
                    }
                );

                const responseText =
                    await response.text();

                return {
                    ok: response.ok,
                    status: response.status,
                    statusText: response.statusText,
                    contentType: (
                        response.headers.get(
                            "content-type"
                        ) || ""
                    ),
                    responseText: responseText
                };
            } catch (error) {
                return {
                    ok: false,
                    status: 0,
                    statusText: "",
                    contentType: "",
                    responseText: "",
                    browserError: (
                        error instanceof Error
                            ? (
                                error.name
                                + ": "
                                + error.message
                            )
                            : String(error)
                    )
                };
            } finally {
                clearTimeout(timeoutHandle);
            }
        }
        """,
        {
            "url": SPATIALANALYZE_STORED_DATA_URL,
            "sessionId": session_id,
            "formValues": formular_data,
            "timeoutMs": timeout_ms,
        },
    )

    fetch_resultat = _valider_fetch_resultat(
        browser_resultat
    )

    browser_error = str(
        fetch_resultat.get("browserError")
        or ""
    ).strip()

    if browser_error:
        raise RuntimeError(
            "WebGIS-rækkerne kunne ikke hentes "
            "i browseren. "
            f"Browserfejl: {browser_error}"
        )

    status = fetch_resultat["status"]

    if not fetch_resultat["ok"]:
        response_preview = (
            _lav_sikkert_response_preview(
                fetch_resultat.get(
                    "responseText"
                )
            )
        )

        raise RuntimeError(
            "Hentning af WebGIS-rækker fejlede med "
            f"HTTP-status {status}. "
            f"Statusbesked: "
            f"{fetch_resultat.get('statusText')!s}. "
            f"Svareksempel: {response_preview}"
        )

    response_text = str(
        fetch_resultat.get("responseText")
        or ""
    ).strip()

    if not response_text:
        raise RuntimeError(
            "WebGIS returnerede et tomt svar ved "
            "hentning af spatialanalyse-rækker."
        )

    resultat = await _parse_json_i_browser(
        page=page,
        response_text=response_text,
    )

    raekker = _find_datarækker(
        resultat
    )

    logger.info(
        "WebGIS returnerede %s læsbare datarækker.",
        len(raekker),
    )

    return raekker


def _find_datarækker(
    value: Any,
) -> list[dict[str, Any]]:
    """Finder den mest sandsynlige liste med datarækker.

    WebGIS kan pakke rækkerne ind i forskellige JSON-strukturer.
    Funktionen gennemgår derfor dictionaries og lister rekursivt.

    Output:
        Den største liste med dictionaries, som ligner datarækker.
    """
    kandidater: list[
        list[dict[str, Any]]
    ] = []

    def besoeg(
        current_value: Any,
    ) -> None:
        if isinstance(
            current_value,
            list,
        ):
            dictionary_rows = [
                element
                for element in current_value
                if isinstance(element, dict)
                and not _er_metadata_raekke(element)
            ]

            if dictionary_rows:
                kandidater.append(
                    dictionary_rows
                )

            for element in current_value:
                besoeg(element)

            return

        if isinstance(
            current_value,
            dict,
        ):
            for nested_value in (
                current_value.values()
            ):
                besoeg(nested_value)

    besoeg(value)

    if not kandidater:
        if (
            isinstance(value, dict)
            and not _er_metadata_raekke(value)
        ):
            return [value]

        return []

    return max(
        kandidater,
        key=len,
    )


def _er_metadata_raekke(
    row: dict[str, Any],
) -> bool:
    """Kontrollerer om en række kun ligner WebGIS-metadata."""
    if not isinstance(row, dict):
        return True

    keys = {
        str(key).casefold()
        for key in row
    }

    metadata_keys = {
        "_name",
        "_class",
        "_pcolid",
        "rowcount",
        "xmin",
        "ymin",
        "xmax",
        "ymax",
    }

    return bool(keys) and keys.issubset(
        metadata_keys
    )

def opret_ticket_listener(
    page: Page,
) -> asyncio.Future[str]:
    """
    Opsnapper Ticket-headeren fra WebGIS' egne requests.

    Listeneren skal oprettes før navigationen.

    Output:
        Future med ticketværdien.

    Sikkerhed:
        Ticketværdien bliver aldrig logget.
    """
    event_loop = asyncio.get_running_loop()

    ticket_future: asyncio.Future[str] = (
        event_loop.create_future()
    )

    async def kontroller_request(
        request: Request,
    ) -> None:
        if ticket_future.done():
            return

        request_hostname = (
            urlparse(request.url).hostname
            or ""
        ).casefold()

        if request_hostname != WEBGIS_HOSTNAME:
            return

        try:
            headers = await request.all_headers()
        except Exception as exc:
            logger.debug(
                "Headers kunne ikke læses fra "
                "WebGIS-request: %s",
                exc,
            )
            return

        ticket = str(
            headers.get(
                TICKET_HEADER_NAME,
                "",
            )
            or ""
        ).strip()

        if (
            ticket
            and not ticket_future.done()
        ):
            logger.info(
                "WebGIS Ticket-headeren er fundet."
            )

            ticket_future.set_result(
                ticket
            )

    def ved_request(
        request: Request,
    ) -> None:
        task = asyncio.create_task(
            kontroller_request(
                request
            )
        )

        task.add_done_callback(
            _log_ticket_listenerfejl
        )

    page.on(
        "request",
        ved_request,
    )

    return ticket_future


def _log_ticket_listenerfejl(
    completed_task: asyncio.Task[None],
) -> None:
    """
    Logger eventuelle fejl fra ticket-listeneren.
    """
    if completed_task.cancelled():
        return

    try:
        exception = completed_task.exception()
    except asyncio.CancelledError:
        return

    if exception is not None:
        logger.error(
            "Fejl i ticket-listeneren: %s",
            exception,
            exc_info=(
                type(exception),
                exception,
                exception.__traceback__,
            ),
        )


async def vent_paa_ticket(
    ticket_future: asyncio.Future[str],
    *,
    timeout_seconds: int = TICKET_TIMEOUT_SECONDS,
) -> str:
    """
    Venter på WebGIS-ticketen.

    Output:
        Ticketværdien.

    Sikkerhed:
        Ticketværdien bliver aldrig logget.
    """
    logger.info(
        "Venter på WebGIS Ticket-headeren."
    )

    try:
        ticket = await asyncio.wait_for(
            asyncio.shield(
                ticket_future
            ),
            timeout=timeout_seconds,
        )
    except TimeoutError as exc:
        raise RuntimeError(
            "WebGIS Ticket-headeren blev ikke "
            f"fundet inden for {timeout_seconds} "
            "sekunder. Åbn bådpladsen og aktivér "
            "redigeringsvisningen."
        ) from exc

    ticket = str(
        ticket
        or ""
    ).strip()

    if not ticket:
        raise RuntimeError(
            "Ticket-listeneren returnerede "
            "en tom ticket."
        )

    return ticket

def opret_session_id_listener(
    page: Page,
) -> asyncio.Future[str]:
    """
    Finder Session-Id i WebGIS' egne requests.

    Listeneren skal oprettes før navigationen til WebGIS.

    Output:
        En Future, som får Session-Id som resultat.

    Sikkerhed:
        Session-Id bliver ikke logget.
    """
    event_loop = asyncio.get_running_loop()

    session_id_future: asyncio.Future[str] = (
        event_loop.create_future()
    )

    async def kontroller_request(
        request: Request,
    ) -> None:
        if session_id_future.done():
            return

        request_hostname = (
            urlparse(request.url).hostname
            or ""
        ).casefold()

        if request_hostname != WEBGIS_HOSTNAME:
            return

        try:
            headers = await request.all_headers()
        except Exception as exc:
            logger.debug(
                "Headers kunne ikke læses fra "
                "WebGIS-request: %s",
                exc,
            )
            return

        session_id = str(
            headers.get(
                SESSION_ID_HEADER_NAME,
                "",
            )
            or ""
        ).strip()

        if not session_id:
            return

        if session_id_future.done():
            return

        logger.info(
            "WebGIS Session-Id er fundet."
        )

        session_id_future.set_result(
            session_id
        )

    def ved_request(
        request: Request,
    ) -> None:
        task = asyncio.create_task(
            kontroller_request(
                request
            )
        )

        task.add_done_callback(
            _log_listenerfejl
        )

    page.on(
        "request",
        ved_request,
    )

    return session_id_future


def _log_listenerfejl(
    completed_task: asyncio.Task[None],
) -> None:
    """
    Logger fejl fra Session-Id-listeneren.
    """
    if completed_task.cancelled():
        return

    try:
        exception = completed_task.exception()
    except asyncio.CancelledError:
        return

    if exception is not None:
        logger.error(
            "Fejl i Session-Id-listeneren: %s",
            exception,
            exc_info=(
                type(exception),
                exception,
                exception.__traceback__,
            ),
        )

async def vent_paa_jsessionid(
    page: Page,
    *,
    timeout_seconds: int = SESSION_TIMEOUT_SECONDS,
) -> None:
    """
    Venter på WebGIS-sessionens JSESSIONID-cookie.

    Cookieværdien bliver ikke returneret eller logget.
    """
    logger.info(
        "Venter på WebGIS-sessionens JSESSIONID."
    )

    event_loop = asyncio.get_running_loop()

    deadline = (
        event_loop.time()
        + timeout_seconds
    )

    while event_loop.time() < deadline:
        cookies = await page.context.cookies(
            [WEBGIS_BASE_URL]
        )

        har_jsessionid = any(
            (
                str(
                    cookie.get(
                        "name",
                        "",
                    )
                ).casefold()
                == JSESSIONID_COOKIE_NAME.casefold()
                and bool(
                    cookie.get("value")
                )
            )
            for cookie in cookies
        )

        if har_jsessionid:
            logger.info(
                "WebGIS-sessionens JSESSIONID "
                "er fundet."
            )
            return

        await asyncio.sleep(
            0.5
        )

    raise RuntimeError(
        "WebGIS blev åbnet, men JSESSIONID blev "
        f"ikke fundet inden for {timeout_seconds} "
        "sekunder."
    )

def find_rowcount(
    value: Any,
) -> int | None:
    """
    Finder det første læsbare rowcount-felt rekursivt.

    Output:
        Rowcount som heltal eller None, hvis feltet ikke findes.
    """
    if isinstance(
        value,
        dict,
    ):
        for key, nested_value in value.items():
            if str(key).casefold() == "rowcount":
                try:
                    rowcount = int(
                        nested_value
                    )
                except (
                    TypeError,
                    ValueError,
                ) as exc:
                    raise ValueError(
                        "Analysens rowcount kunne ikke "
                        "konverteres til et heltal."
                    ) from exc

                return rowcount

        for nested_value in value.values():
            found_rowcount = find_rowcount(
                nested_value
            )

            if found_rowcount is not None:
                return found_rowcount

    elif isinstance(
        value,
        list,
    ):
        for nested_value in value:
            found_rowcount = find_rowcount(
                nested_value
            )

            if found_rowcount is not None:
                return found_rowcount

    return None


def valider_analyseresultat(
    analyseresultat: dict[str, Any],
) -> int:
    """
    Validerer spatialanalyze-resultatet.

    Output:
        Analysens rowcount som heltal.
    """
    if not isinstance(
        analyseresultat,
        dict,
    ):
        raise TypeError(
            "spatialanalyze() skal returnere "
            "en dictionary."
        )

    if not analyseresultat:
        raise AssertionError(
            "spatialanalyze() returnerede et "
            "tomt resultat."
        )

    rowcount = find_rowcount(
        analyseresultat
    )

    if rowcount is None:
        raise AssertionError(
            "Analysesvaret indeholder ikke et "
            "læsbart rowcount-felt."
        )

    if rowcount < 0:
        raise AssertionError(
            "Analysens rowcount må ikke være negativ. "
            f"Rowcount: {rowcount}."
        )

    if rowcount > BAADPLADS_MAXROWS:
        raise AssertionError(
            "Analysens rowcount er større end "
            "BAADPLADS_MAXROWS. "
            f"Rowcount: {rowcount}. "
            f"Maxrows: {BAADPLADS_MAXROWS}."
        )

    if (
        rowcount
        != TIDLIGERE_OBSERVERET_ROWCOUNT
    ):
        logger.warning(
            "Analysens rowcount er ændret. "
            "Tidligere observeret: %s. "
            "Aktuelt resultat: %s.",
            TIDLIGERE_OBSERVERET_ROWCOUNT,
            rowcount,
        )

    return rowcount


def valider_baadpladser(
    baadpladser: list[dict[str, Any]],
    *,
    forventet_antal: int,
) -> None:
    """
    Validerer de hentede bådpladsrækker.

    Funktionen kontrollerer både datatypen og antallet af rækker.
    """
    if not isinstance(
        baadpladser,
        list,
    ):
        raise TypeError(
            "hent_baadpladser() skal returnere "
            "en liste."
        )

    for index, baadplads in enumerate(
        baadpladser,
        start=1,
    ):
        if not isinstance(
            baadplads,
            dict,
        ):
            raise TypeError(
                "Alle bådpladsrækker skal være "
                "dictionaries. "
                f"Række {index} har typen "
                f"{type(baadplads).__name__}."
            )

    if len(baadpladser) != forventet_antal:
        raise AssertionError(
            "Antallet af hentede bådpladsrækker "
            "svarer ikke til analysens rowcount. "
            f"Rowcount: {forventet_antal}. "
            f"Hentede rækker: {len(baadpladser)}."
        )

def _hent_queue_id(workqueue: Any) -> int:
    """Returnerer det validerede tekniske queue-id."""
    queue_id = getattr(workqueue, "id", None)

    if queue_id is None or isinstance(queue_id, bool):
        raise ValueError(
            "Automation Server-workqueuen mangler et gyldigt id."
        )

    try:
        validated_queue_id = int(queue_id)
    except (TypeError, ValueError) as exc:
        raise ValueError(
            "workqueue.id skal være et heltal."
        ) from exc

    if validated_queue_id <= 0:
        raise ValueError(
            "workqueue.id skal være større end 0."
        )

    return validated_queue_id


def _hent_item_data_fra_ats(
    *,
    queue_id: int,
    item_reference: str,
) -> dict[str, Any]:
    """Henter itemets data læsende via box.caseid."""
    result = get_workitems_by_box_value(
        queue_id=queue_id,
        box_path="caseid",
        value=item_reference,
        statuses=None,
        limit=10,
    )
    items = result.get("items")

    if not isinstance(items, list):
        raise RuntimeError(
            "ATS-opslaget returnerede ikke items som en liste."
        )

    matches = [
        row
        for row in items
        if isinstance(row, dict)
        and str(row.get("reference") or "").strip()
        == item_reference
    ]

    if not matches:
        raise RuntimeError(
            "Ingen konfliktrapport-items blev fundet med reference "
            f"{item_reference!r} i queue {queue_id}."
        )

    if len(matches) > 1:
        item_ids = [row.get("id") for row in matches]
        raise RuntimeError(
            "Der blev fundet flere items med samme reference. "
            f"Item-id'er: {item_ids}."
        )

    database_row = matches[0]
    data = database_row.get("data")

    if not isinstance(data, dict):
        raise RuntimeError(
            "Det fundne items data er ikke en dictionary."
        )

    logger.info(
        "ATS-item fundet læsende: id=%s, reference=%s, status=%s.",
        database_row.get("id"),
        database_row.get("reference"),
        database_row.get("status"),
    )

    return data


def _hent_webgis_data(
    item_data: dict[str, Any],
) -> dict[str, Any]:
    """Returnerer item.data['box']['webgis']."""
    box = item_data.get("box")

    if not isinstance(box, dict):
        raise RuntimeError(
            "Itemet mangler data['box']."
        )

    webgis_data = box.get("webgis")

    if not isinstance(webgis_data, dict):
        raise RuntimeError(
            "Itemet mangler data['box']['webgis']."
        )

    return webgis_data


def _hent_webgis_http_credentials() -> dict[str, str]:
    """Henter WebGIS HTTP-login fra Automation Server."""
    credential = Credential.get_credential(
        WEBGIS_CREDENTIAL_NAME
    )

    if credential is None:
        raise RuntimeError(
            f"Credentialen {WEBGIS_CREDENTIAL_NAME!r} "
            "blev ikke fundet."
        )

    username = str(credential.username or "").strip()
    password = str(credential.password or "")

    if not username or not password:
        raise RuntimeError(
            f"Credentialen {WEBGIS_CREDENTIAL_NAME!r} "
            "mangler username eller password."
        )

    return {
        "username": username,
        "password": password,
    }


def _vaelg_webgis_tema(
    buildingregulation: str,
) -> str:
    """Vælger samme WebGIS-tema som konfliktrapport-processen."""
    normaliseret = " ".join(
        buildingregulation.split()
    ).casefold()

    if "landzonesag" in normaliseret:
        return "landzonesager"

    return "byggesag"


def _paakraevet_tekst(
    value: Any,
    field_name: str,
) -> str:
    """Returnerer en obligatorisk, trimmet tekstværdi."""
    text = str(value or "").strip()

    if not text:
        raise ValueError(
            f"{field_name} mangler eller er tom."
        )

    return text


def _valider_pdf(rapportsti: Path) -> None:
    """Kontrollerer eksistens, størrelse og PDF-signatur."""
    if not rapportsti.is_file():
        raise AssertionError(
            "PDF-rapporten blev ikke fundet: "
            f"{rapportsti}"
        )

    if rapportsti.stat().st_size == 0:
        raise AssertionError(
            "PDF-rapporten er tom."
        )

    with rapportsti.open("rb") as pdf_file:
        if pdf_file.read(5) != b"%PDF-":
            raise AssertionError(
                "Rapportfilen har ikke en gyldig PDF-signatur."
            )

async def vent_paa_jsessionid(
    page: Page,
    timeout_seconds: int = COOKIE_TIMEOUT_SECONDS,
) -> bool:
    """Venter på JSESSIONID i den aktuelle browserkontekst.

    Output:
        True, når JSESSIONID findes.

    Sikkerhed:
        Cookieværdien bliver aldrig udskrevet eller returneret.
    """
    antal_forsog = timeout_seconds * 2

    for _ in range(antal_forsog):
        cookies = await page.context.cookies(
            [WEBGIS_URL]
        )

        har_jsessionid = any(
            cookie.get("name", "").casefold()
            == JSESSIONID_COOKIE_NAME.casefold()
            and bool(cookie.get("value"))
            for cookie in cookies
        )

        if har_jsessionid:
            logger.info(
                "WebGIS-sessionens JSESSIONID er fundet."
            )
            return True

        await asyncio.sleep(0.5)

    raise RuntimeError(
        "WebGIS blev åbnet, men JSESSIONID blev ikke "
        f"fundet inden for {timeout_seconds} sekunder."
    )


async def udskriv_cookie_navne(

    cookie_names = sorted(
        {
            str(cookie.get("name", "")).strip()
            for cookie in cookies
            if cookie.get("name")
        }
    )

    logger.info(
        "Cookienavne efter WebGIS-login: %s",
        cookie_names,
    )

def opret_session_id_listener(
    page: Page,
) -> asyncio.Future:
    event_loop = asyncio.get_running_loop()
    session_id_future: asyncio.Future[str] = event_loop.create_future()

    async def kontroller_request(request: Request) -> None:
        if session_id_future.done():
            return

        request_hostname = (urlparse(request.url).hostname or "").casefold()

        if request_hostname != WEBGIS_HOSTNAME:
            return

        try:
            headers = await request.all_headers()
        except Exception as exc:
            logger.debug(
                "Headers kunne ikke læses fra et WebGIS-request: %s",
                exc,
            )
            return

        session_id = str(
            headers.get(SESSION_ID_HEADER_NAME, "") or ""
        ).strip()

        if not session_id or session_id_future.done():
            return

        logger.info("WebGIS Session-Id er fundet automatisk.")
        session_id_future.set_result(session_id)

    def log_listenerfejl(completed_task: asyncio.Task[None]) -> None:
        if completed_task.cancelled():
            return

        try:
            exception = completed_task.exception()
        except asyncio.CancelledError:
            return

        if exception is not None:
            logger.error(
                "Fejl i Session-Id-listeneren: %s",
                exception,
                exc_info=(
                    type(exception),
                    exception,
                    exception.__traceback__,
                ),
            )

    def ved_request(request: Request) -> None:
        task = asyncio.create_task(kontroller_request(request))
        task.add_done_callback(log_listenerfejl)

    page.on("request", ved_request)
    return session_id_future


async def vent_paa_session_id(
    session_id_future: asyncio.Future[str],
    timeout_seconds: int = SESSION_TIMEOUT_SECONDS,
) -> str:
    """Venter på Session-Id fra WebGIS' egne requests."""
    logger.info("Venter på WebGIS Session-Id.")

    try:
        session_id = await asyncio.wait_for(
            asyncio.shield(session_id_future),
            timeout=timeout_seconds,
        )
    except TimeoutError as exc:
        raise RuntimeError(
            "WebGIS blev åbnet, men Session-Id blev ikke fundet "
            f"inden for {timeout_seconds} sekunder."
        ) from exc

    session_id = str(session_id or "").strip()

    if not session_id:
        raise RuntimeError(
            "WebGIS-listeneren returnerede en tom Session-Id."
        )

    return session_id

async def vent_paa_jsessionid(
    page: Page,
    timeout_seconds: int = SESSION_TIMEOUT_SECONDS,
) -> None:
    """Venter på JSESSIONID uden at returnere cookieindholdet."""
    logger.info("Venter på WebGIS-sessionens JSESSIONID.")

    event_loop = asyncio.get_running_loop()
    deadline = event_loop.time() + timeout_seconds

    while event_loop.time() < deadline:
        cookies = await page.context.cookies([WEBGIS_BASE_URL])

        har_jsessionid = any(
            str(cookie.get("name", "")).casefold()
            == JSESSIONID_COOKIE_NAME.casefold()
            and bool(cookie.get("value"))
            for cookie in cookies
        )

        if har_jsessionid:
            logger.info("WebGIS-sessionens JSESSIONID er fundet.")
            return

        await asyncio.sleep(0.5)

    raise RuntimeError(
        "WebGIS blev åbnet, men JSESSIONID blev ikke fundet "
        f"inden for {timeout_seconds} sekunder."
    )

def find_rowcount(value: Any) -> int | None:
    """Finder rowcount rekursivt i analysesvaret."""
    if isinstance(value, dict):
        for key, nested_value in value.items():
            if str(key).casefold() == "rowcount":
                try:
                    return int(nested_value)
                except (TypeError, ValueError) as exc:
                    raise ValueError(
                        "Analysens rowcount kunne ikke konverteres "
                        "til et heltal."
                    ) from exc

        for nested_value in value.values():
            found_rowcount = find_rowcount(nested_value)
            if found_rowcount is not None:
                return found_rowcount

    elif isinstance(value, list):
        for nested_value in value:
            found_rowcount = find_rowcount(nested_value)
            if found_rowcount is not None:
                return found_rowcount

    return None


def valider_antal_raekker(
    resultat: dict[str, Any],
    raekker: list[dict[str, Any]],
) -> int | None:
    """Sammenligner analysens rowcount med de hentede rækker."""
    rowcount = find_rowcount(resultat)

    if rowcount is None:
        logger.warning(
            "Analysesvaret indeholder ikke et læsbart rowcount-felt."
        )
        return None

    if rowcount < 0:
        raise AssertionError(
            f"Analysens rowcount må ikke være negativ: {rowcount}."
        )

    if rowcount > MAX_ROWS:
        raise AssertionError(
            "Analysens rowcount er større end MAX_ROWS. "
            f"Rowcount: {rowcount}. MAX_ROWS: {MAX_ROWS}."
        )

    if len(raekker) != rowcount:
        raise AssertionError(
            "Antallet af hentede rækker svarer ikke til analysens "
            f"rowcount. Rowcount: {rowcount}. "
            f"Hentede rækker: {len(raekker)}."
        )

    logger.info(
        "Antal hentede rækker matcher rowcount: %s.",
        rowcount,
    )
    return rowcount

_DOWNLOADS_BY_PAGE: dict[int, Download] = {}


async def _aabn_rapport_med_download(
    page: Page,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Page:
    """Klikker Vis rapport og fanger den faktiske Playwright-download."""
    rapport_knap = page.locator(RAPPORT_KNAP).filter(
        has_text=re.compile(r"^\s*Rapport\s*$", re.IGNORECASE)
    ).first
    await rapport_knap.wait_for(state="visible", timeout=timeout_ms)
    await rapport_knap.click()
    await page.wait_for_timeout(1_000)

    vis_rapport_knap = page.locator(VIS_RAPPORT_KNAP).filter(
        has_text=re.compile(r"^\s*Vis rapport\s*$", re.IGNORECASE)
    ).first
    await vis_rapport_knap.wait_for(state="visible", timeout=timeout_ms)

    loop = asyncio.get_running_loop()
    download_future: asyncio.Future[Download] = loop.create_future()

    def ved_download(download: Download) -> None:
        if not download_future.done():
            download_future.set_result(download)

    def ved_ny_side(ny_side: Page) -> None:
        ny_side.on("download", ved_download)

    page.on("download", ved_download)
    page.context.on("page", ved_ny_side)

    logger.info("Bestiller rapporten og venter på download-event.")

    try:
        async with page.context.expect_page(timeout=timeout_ms) as popup_info:
            await vis_rapport_knap.click()

        rapport_side = await popup_info.value
        rapport_side.on("download", ved_download)

        download = await asyncio.wait_for(
            asyncio.shield(download_future),
            timeout=timeout_ms / 1_000,
        )

        fejl = await download.failure()
        if fejl:
            raise RuntimeError(f"WebGIS-download fejlede: {fejl}")

        _DOWNLOADS_BY_PAGE[id(rapport_side)] = download

        logger.info(
            "Download-event modtaget. Foreslået filnavn: %s",
            download.suggested_filename,
        )
        return rapport_side
    finally:
        page.remove_listener("download", ved_download)

async def _aabn_rapport_headless(
    page: Page,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Page:
    """Åbner rapporten og opsnapper PDF-responsens URL fra netværket."""
    rapport_knap = page.locator(RAPPORT_KNAP).filter(
        has_text=re.compile(r"^\s*Rapport\s*$", re.IGNORECASE)
    ).first
    await rapport_knap.wait_for(state="visible", timeout=timeout_ms)

    logger.info("Åbner WebGIS-rapportdialogen.")
    await rapport_knap.click()
    await page.wait_for_timeout(1_000)

    vis_rapport_knap = page.locator(VIS_RAPPORT_KNAP).filter(
        has_text=re.compile(r"^\s*Vis rapport\s*$", re.IGNORECASE)
    ).first
    await vis_rapport_knap.wait_for(state="visible", timeout=timeout_ms)

    pdf_url_future: asyncio.Future[str] = (
        asyncio.get_running_loop().create_future()
    )

    def ved_response(response: Response) -> None:
        if pdf_url_future.done():
            return

        content_type = str(
            response.headers.get("content-type", "")
        ).casefold()
        content_disposition = str(
            response.headers.get("content-disposition", "")
        ).casefold()
        response_url = str(response.url or "").strip()

        er_pdf = (
            "application/pdf" in content_type
            or ".pdf" in response_url.casefold()
            or "filename=" in content_disposition
            and ".pdf" in content_disposition
        )

        if er_pdf and response_url:
            logger.info(
                "PDF-respons opsnappet: content-type=%s",
                content_type or "ukendt",
            )
            pdf_url_future.set_result(response_url)

    page.context.on("response", ved_response)

    logger.info("Bestiller den færdige PDF-rapport.")

    try:
        async with page.context.expect_page(
            timeout=timeout_ms,
        ) as ny_side_info:
            await vis_rapport_knap.click()

        rapport_side = await ny_side_info.value

        pdf_url = await asyncio.wait_for(
            asyncio.shield(pdf_url_future),
            timeout=timeout_ms / 1_000,
        )

        _PDF_URL_BY_PAGE[id(rapport_side)] = pdf_url
        logger.info("PDF-URL gemt fra netværksresponsen.")

        loading = page.locator(RAPPORT_LOADING).first
        if await loading.count() > 0:
            try:
                await loading.wait_for(
                    state="hidden",
                    timeout=timeout_ms,
                )
            except Exception:
                logger.debug(
                    "Rapport-loading kunne ikke længere findes."
                )

        return rapport_side
    finally:
        page.context.remove_listener("response", ved_response)

def parse_wkt_polygon(
    wkt: str,
) -> list[tuple[float, float]]:
    """
    Parser en enkel WKT POLYGON-værdi.

    Output:
        Liste med polygonpunkter som (x, y).
    """

    if not isinstance(wkt, str):
        raise TypeError(
            "wkt skal være tekst."
        )

    text = wkt.strip()

    if not text:
        raise ValueError(
            "wkt må ikke være tom."
        )

    start = "POLYGON (("
    slut = "))"

    if not text.upper().startswith(start):
        raise ValueError(
            "Kun en enkel WKT POLYGON understøttes."
        )

    if not text.endswith(slut):
        raise ValueError(
            "WKT-polygonen mangler afsluttende parenteser."
        )

    koordinattekst = text[
        len(start):-len(slut)
    ]

    points: list[tuple[float, float]] = []

    for indeks, pair in enumerate(
        koordinattekst.split(","),
        start=1,
    ):
        dele = pair.strip().split()

        if len(dele) != 2:
            raise ValueError(
                "WKT-koordinatparret på position "
                f"{indeks} indeholder ikke præcis x og y."
            )

        try:
            x = float(dele[0])
            y = float(dele[1])
        except ValueError as exc:
            raise ValueError(
                "WKT-koordinatparret på position "
                f"{indeks} indeholder ugyldige tal."
            ) from exc

        points.append(
            (
                x,
                y,
            )
        )

    if len(points) < 4:
        raise ValueError(
            "En polygon skal indeholde mindst fire punkter."
        )

    if points[0] != points[-1]:
        raise ValueError(
            "WKT-polygonens første og sidste punkt skal være ens."
        )

    return points


def calculate_bbox(
    points: list[tuple[float, float]],
    buffer_meter: float = 50.0,
) -> tuple[float, float, float, float]:
    """
    Beregner polygonens bounding box med buffer.

    Output:
        xmin, ymin, xmax, ymax.
    """

    if not points:
        raise ValueError(
            "points må ikke være tom."
        )

    if isinstance(buffer_meter, bool):
        raise TypeError(
            "buffer_meter skal være et tal."
        )

    try:
        buffer_vaerdi = float(
            buffer_meter
        )
    except (TypeError, ValueError) as exc:
        raise TypeError(
            "buffer_meter skal være et tal."
        ) from exc

    if buffer_vaerdi < 0:
        raise ValueError(
            "buffer_meter må ikke være negativ."
        )

    xs = [x for x, _ in points]
    ys = [y for _, y in points]

    xmin = min(xs) - buffer_vaerdi
    ymin = min(ys) - buffer_vaerdi
    xmax = max(xs) + buffer_vaerdi
    ymax = max(ys) + buffer_vaerdi

    if xmin >= xmax or ymin >= ymax:
        raise ValueError(
            "Den beregnede bounding box er ugyldig."
        )

    return (
        xmin,
        ymin,
        xmax,
        ymax,
    )


def world_to_pixel(
    x: float,
    y: float,
    bbox: tuple[float, float, float, float],
    width: int,
    height: int,
) -> tuple[int, int]:
    """
    Konverterer en koordinat til en pixelposition.

    Output:
        Pixelpositionen som (x, y).
    """

    valider_billedstoerrelse(
        width,
        height,
    )

    xmin, ymin, xmax, ymax = bbox

    if xmin >= xmax or ymin >= ymax:
        raise ValueError(
            "bbox er ugyldig."
        )

    px = (
        (x - xmin)
        / (xmax - xmin)
        * width
    )

    py = height - (
        (y - ymin)
        / (ymax - ymin)
        * height
    )

    return (
        round(px),
        round(py),
    )

def hent_kort(
    bbox: tuple[float, float, float, float],
    *,
    token: str,
    width: int = WIDTH,
    height: int = HEIGHT,
    timeout_seconds: int = REQUEST_TIMEOUT_SECONDS,
) -> Image.Image:
    """
    Henter et baggrundskort fra Dataforsyningens WMS.

    Output:
        Et færdigindlæst Pillow-billede i RGBA-format.
    """

    if not isinstance(token, str) or not token.strip():
        raise ValueError(
            "token skal være en ikke-tom tekst."
        )

    valider_billedstoerrelse(
        width,
        height,
    )

    if isinstance(timeout_seconds, bool) or not isinstance(
        timeout_seconds,
        int,
    ):
        raise TypeError(
            "timeout_seconds skal være et heltal."
        )

    if timeout_seconds <= 0:
        raise ValueError(
            "timeout_seconds skal være større end 0."
        )

    xmin, ymin, xmax, ymax = bbox

    params = {
        "SERVICE": "WMS",
        "VERSION": "1.1.1",
        "REQUEST": "GetMap",
        "FORMAT": "image/png",
        "TRANSPARENT": "TRUE",
        "LAYERS": "Basis_kort",
        "token": token.strip(),
        "servicename": "forvaltning2",
        "SRS": "EPSG:25832",
        "STYLES": "",
        "WIDTH": width,
        "HEIGHT": height,
        "BBOX": f"{xmin},{ymin},{xmax},{ymax}",
    }

    response = requests.get(
        DATAFORSYNING_WMS_URL,
        params=params,
        timeout=timeout_seconds,
    )

    response.raise_for_status()

    content_type = str(
        response.headers.get(
            "content-type",
            "",
        )
    ).casefold()

    if "image/" not in content_type:
        raise RuntimeError(
            "Dataforsyningen returnerede ikke et billede. "
            f"Content-Type var {content_type!r}."
        )

    try:
        with Image.open(
            BytesIO(response.content)
        ) as source_image:
            source_image.load()

            return source_image.convert(
                "RGBA"
            ).copy()

    except Exception as exc:
        raise RuntimeError(
            "Kortsvaret kunne ikke åbnes som et billede."
        ) from exc

    )

    unik_reference = uuid4().hex
    grundnavn = f"kort_med_polygon_{unik_reference}"

    return (
        mappe / f"{grundnavn}.png",
        mappe / f"{grundnavn}.pdf",
    )


def gem_kortfiler(
    image: Image.Image,
    points: list[tuple[float, float]],
    bbox: tuple[float, float, float, float],
    *,
    output_mappe: str | Path = OUTPUT_MAPPE,
) -> tuple[Path, Path]:
    """
    Tegner polygonen og gemmer unikke PNG- og PDF-filer.

    Output:
        Tuple med den gemte PNG-sti og PDF-sti.
    """

    if image is None:
        raise ValueError(
            "image skal angives."
        )

    pixel_points = [
        world_to_pixel(
            x=x,
            y=y,
            bbox=bbox,
            width=image.width,
            height=image.height,
        )
        for x, y in points
    ]

    output_image = image.copy()

    draw = ImageDraw.Draw(
        output_image
    )

    draw.line(
        pixel_points,
        fill="red",
        width=POLYGON_LINE_WIDTH,
        joint="curve",
    )

    png_sti, pdf_sti = opret_unikke_outputstier(
        output_mappe
    )

    output_image.save(
        png_sti,
        format="PNG",
    )

    output_image.convert(
        "RGB"
    ).save(
        pdf_sti,
        format="PDF",
        resolution=PDF_RESOLUTION,
    )

    if not png_sti.exists() or png_sti.stat().st_size == 0:
        raise RuntimeError(
            "PNG-filen blev ikke gemt korrekt."
        )

    if not pdf_sti.exists() or pdf_sti.stat().st_size == 0:
        raise RuntimeError(
            "PDF-filen blev ikke gemt korrekt."
        )

    if not pdf_sti.read_bytes().startswith(
        b"%PDF-"
    ):
        raise RuntimeError(
            "Den gemte PDF mangler en gyldig PDF-signatur."
        )

    return (
        png_sti,
        pdf_sti,
    )

def main() -> None:
    """
    Henter et kort, tegner WKT-polygonen og gemmer unikke filer.

    Output:
        Gemmer én PNG og én PDF og udskriver deres absolutte stier.
    """

    load_dotenv()

    logging.basicConfig(
        level=logging.INFO,
        format=(
            "%(asctime)s [%(levelname)s] "
            "%(name)s: %(message)s"
        ),
        force=True,
    )

    token = hent_dataforsyning_token()

    points = parse_wkt_polygon(
        WKT
    )

    bbox = calculate_bbox(
        points,
        buffer_meter=BBOX_BUFFER_METER,
    )

    logger.info(
        "BBOX: %s",
        bbox,
    )

    image = hent_kort(
        bbox,
        token=token,
    )

    png_sti, pdf_sti = gem_kortfiler(
        image=image,
        points=points,
        bbox=bbox,
    )

    logger.info(
        "PNG gemt: %s",
        png_sti.resolve(),
    )

    logger.info(
        "PDF gemt: %s",
        pdf_sti.resolve(),
    )


if __name__ == "__main__":
    main()

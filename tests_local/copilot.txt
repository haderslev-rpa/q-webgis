async def opret_webgis_rapport(
    page: Page,
    session: BrowserSession,
    wkt: str,
    *,
    tema: str = WEBGIS_STANDARD_TEMA,
    rapportmappe: str | Path = WEBGIS_RAPPORT_MAPPE,
    filnavn: str = WEBGIS_RAPPORT_FILNAVN,
    vaelg_jordstykker_byggesag: bool = True,
    timeout_ms: int = WEBGIS_TIMEOUT_MS,
    rapport_timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Path:
    """
    Udfører hele WebGIS-flowet og gemmer rapporten som PDF.

    BrowserSession modtages fra testen eller den kaldende proces.
    Funktionen opretter eller lukker ikke selve BrowserSession.

    Flow:
        1. Tager screenshot af WebGIS-startsiden.
        2. Vælger eventuelt temaet "Jordstykker - Byggesag".
        3. Åbner "Hvad gælder for" og indsætter WKT.
        4. Vælger og udfører konfliktsøgningen.
        5. Venter på status "Færdig".
        6. Åbner rapportdialogen.
        7. Klikker "Vis rapport".
        8. Henter PDF'en gennem den autentificerede WebGIS-side.
        9. Tager screenshots efter hvert hovedtrin.

    Output:
        Path til den gemte PDF-fil.
    """

    _valider_page(
        page
    )

    if session is None:
        raise ValueError(
            "session skal angives."
        )

    valideret_wkt = _valider_wkt(
        wkt
    )

    valideret_tema = _valider_tema(
        tema
    )

    valideret_filnavn = _valider_pdf_filnavn(
        filnavn
    )

    destination = _opret_rapportsti(
        rapportmappe=rapportmappe,
        filnavn=valideret_filnavn,
    )

    rapport_side: Page | None = None

    try:
        logger.info(
            "Starter WebGIS-rapport for konfliktsøgning=%s.",
            valideret_tema,
        )

        await _tag_screenshot(
            session=session,
            page=page,
            navn="STEP_01_webgis_start",
        )

        if vaelg_jordstykker_byggesag:
            await vaelg_jordstykker_byggesag_i_webgis(
                page=page,
                timeout_ms=timeout_ms,
            )

            await _tag_screenshot(
                session=session,
                page=page,
                navn="STEP_02_jordstykker_byggesag_valgt",
            )

        await indsaet_wkt_i_webgis(
            page=page,
            wkt=valideret_wkt,
            timeout_ms=timeout_ms,
        )

        await _tag_screenshot(
            session=session,
            page=page,
            navn="STEP_03_wkt_indsaet",
        )

        await udfoer_konfliktsoegning(
            page=page,
            tema=valideret_tema,
            timeout_ms=rapport_timeout_ms,
        )

        await _tag_screenshot(
            session=session,
            page=page,
            navn="STEP_04_konfliktsoegning_faerdig",
        )

        rapport_side = await aabn_rapport(
            page=page,
            timeout_ms=rapport_timeout_ms,
        )

        await _tag_screenshot(
            session=session,
            page=rapport_side,
            navn="STEP_05_rapport_aabnet",
        )


        pdf_sti = await gem_rapport_pdf(
            webgis_side=page,
            rapport_side=rapport_side,
            destination=destination,
            timeout_ms=rapport_timeout_ms,
        )

        await _tag_screenshot(
            session=session,
            page=rapport_side,
            navn="STEP_06_pdf_rapport_gemt",
        )

        logger.info(
            "WebGIS-rapporten er gemt: %s",
            pdf_sti,
        )

        return pdf_sti

    except Exception:
        logger.exception(
            "WebGIS-rapportflowet fejlede."
        )

        screenshot_side = (
            rapport_side
            if (
                rapport_side is not None
                and not rapport_side.is_closed()
            )
            else page
        )

        await _tag_screenshot(
            session=session,
            page=screenshot_side,
            navn="FEJL_webgis_rapport",
        )

        raise

    finally:
        if (
            rapport_side is not None
            and rapport_side != page
            and not rapport_side.is_closed()
        ):
            await rapport_side.close()

async def vaelg_jordstykker_byggesag_i_webgis(
    page: Page,
    *,
    timeout_ms: int = WEBGIS_TIMEOUT_MS,
) -> None:
    """
    Søger efter temaet "Jordstykker - Byggesag".

    Efter søgeteksten er indsat, klikkes der igen i søgefeltet.
    Derefter klikkes der på søgeresultatgruppen "Temaer".

    Output:
        Funktionen returnerer ikke data.
        Temaresultatet bliver åbnet i WebGIS.
    """

    soegning = "Jordstykker - Byggesag"

    logger.info(
        "Søger efter temaet %s.",
        soegning,
    )

    soegefelt = page.locator(
        GLOBALT_SOEGEFELT
    ).first

    await _vent_paa_synlig(
        soegefelt,
        navn="globalt søgefelt",
        timeout_ms=timeout_ms,
    )

    await soegefelt.click()

    await _vent_mellem_selector_actions(
        page
    )

    await soegefelt.fill(
        soegning
    )

    await _vent_mellem_selector_actions(
        page
    )

    logger.info(
        "Søgeteksten er indsat. "
        "Klikker i søgefeltet igen."
    )

    await soegefelt.click()

    await _vent_mellem_selector_actions(
        page
    )

    tema_resultat = page.locator(
        TEMAER_RESULTAT
    ).first

    await _vent_paa_synlig(
        tema_resultat,
        navn='søgeresultatgruppen "Temaer"',
        timeout_ms=timeout_ms,
    )

    logger.info(
        'Søgeresultatgruppen "Temaer" er synlig. '
        "Klikker på resultatet."
    )

    await tema_resultat.click()

    await _vent_mellem_selector_actions(
        page
    )

    logger.info(
        'Søgeresultatgruppen "Temaer" er valgt '
        'efter søgning efter %s.',
        soegning,
    )

async def indsaet_wkt_i_webgis(
    page: Page,
    wkt: str,
    *,
    timeout_ms: int = WEBGIS_TIMEOUT_MS,
) -> None:
    """
    Åbner arbejdsgeometrien og indsætter WKT.

    Output:
        Funktionen returnerer ikke data. Geometrien indsættes i kortet.
    """

    valideret_wkt = _valider_wkt(wkt)

    logger.info(
        "Åbner WebGIS-arbejdsgeometrien."
    )

    arbejdsgeometri = page.locator(
        HVAD_GAELDER_FOR_MENU
    ).first

    await _vent_paa_synlig(
        arbejdsgeometri,
        navn="Hvad gælder for-menu",
        timeout_ms=timeout_ms,
    )

    # Klik på linket inde i li-elementet, hvis det findes.
    arbejdsgeometri_link = arbejdsgeometri.locator(
        'a[title="Hvad gælder for"]'
    )

    if await arbejdsgeometri_link.count() > 0:
        await arbejdsgeometri_link.click()

        await _vent_mellem_selector_actions(
            page
        )
    else:
        await arbejdsgeometri.click()

        await _vent_mellem_selector_actions(
            page
        )

    wkt_knap = page.locator(
        INDSAET_WKT_KNAP
    ).first

    await _vent_paa_synlig(
        wkt_knap,
        navn="Indsæt WKT i kortet",
        timeout_ms=timeout_ms,
    )

    await wkt_knap.click()

    await _vent_mellem_selector_actions(
        page
    )

    wkt_felt = page.locator(
        WKT_TEKSTFELT
    ).first

    await _vent_paa_synlig(
        wkt_felt,
        navn="WKT-tekstfelt",
        timeout_ms=timeout_ms,
    )

    await wkt_felt.fill(
        valideret_wkt
    )

    await _vent_mellem_selector_actions(
        page
    )

    if await wkt_felt.input_value() != valideret_wkt:
        raise RuntimeError(
            "WKT-tekstfeltet indeholder ikke den forventede WKT."
        )

    ok_knap = page.locator(
        WKT_OK_KNAP
    ).first

    await _vent_paa_aktiv(
        ok_knap,
        navn="OK-knappen til WKT",
        timeout_ms=timeout_ms,
    )

    await ok_knap.click()

    await _vent_mellem_selector_actions(
        page
    )

    hvad_gaelder_for = page.locator(
        HVAD_GAELDER_FOR_KNAP
    ).first

    await _vent_paa_aktiv(
        hvad_gaelder_for,
        navn="Hvad gælder for-knappen",
        timeout_ms=timeout_ms,
    )

    logger.info(
        "WKT er indsat i WebGIS."
    )

async def udfoer_konfliktsoegning(
    page: Page,
    *,
    tema: str,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> None:
    """
    Vælger en konfliktsøgning og venter på WebGIS-status "Færdig".
    """

    valideret_tema = _valider_tema(tema)

    hvad_gaelder_for = page.locator(
        HVAD_GAELDER_FOR_KNAP
    ).first

    await _vent_paa_aktiv(
        hvad_gaelder_for,
        navn="Hvad gælder for-knappen",
        timeout_ms=timeout_ms,
    )

    logger.info(
        "Åbner konfliktsøgningen."
    )

    await hvad_gaelder_for.click()

    await _vent_mellem_selector_actions(
        page
    )

    dropdown = page.locator(
        KONFLIKTSOEGNING_DROPDOWN
    ).first

    await _vent_paa_synlig(
        dropdown,
        navn="dropdown til konfliktsøgning",
        timeout_ms=timeout_ms,
    )

    await dropdown.select_option(
        value=valideret_tema
    )

    await _vent_mellem_selector_actions(
        page
    )

    valgt_tema = await dropdown.input_value()

    if valgt_tema != valideret_tema:
        raise RuntimeError(
            "WebGIS valgte ikke den forventede "
            f"konfliktsøgning. Forventede {valideret_tema!r}, "
            f"men fik {valgt_tema!r}."
        )

    udfoer_knap = page.locator(
        KONFLIKTSOEGNING_UDFOER_KNAP
    ).first

    await _vent_paa_aktiv(
        udfoer_knap,
        navn="Udfør-knappen",
        timeout_ms=timeout_ms,
    )

    logger.info(
        "Starter konfliktsøgning=%s.",
        valideret_tema,
    )

    await udfoer_knap.click()

    await _vent_mellem_selector_actions(
        page
    )

    await _vent_paa_konfliktsoegning_faerdig(
        page=page,
        timeout_ms=timeout_ms,
    )

    logger.info(
        "Konfliktsøgningen er færdig."
    )


async def _vent_paa_konfliktsoegning_faerdig(
    page: Page,
    *,
    timeout_ms: int,
) -> None:
    """
    Venter på status "Færdig" og kontrollerer, at loading-logoet er væk.
    """

    status = page.locator(
        KONFLIKTSOEGNING_STATUS
    ).first

    try:
        await status.wait_for(
            state="visible",
            timeout=timeout_ms,
        )

        await status.filter(
            has_text=re.compile(
                r"^\s*Færdig\s*$",
                re.IGNORECASE,
            )
        ).wait_for(
            state="visible",
            timeout=timeout_ms,
        )
    except PlaywrightTimeoutError as exc:
        status_tekst = ""

        if await status.count() > 0:
            status_tekst = (
                await status.text_content()
                or ""
            ).strip()

        raise RuntimeError(
            "WebGIS-konfliktsøgningen blev ikke færdig. "
            f"Seneste synlige status: {status_tekst!r}."
        ) from exc

    loading = page.locator(
        KONFLIKTSOEGNING_LOADING
    ).first

    if await loading.count() > 0:
        try:
            await loading.wait_for(
                state="hidden",
                timeout=timeout_ms,
            )
        except PlaywrightTimeoutError as exc:
            raise RuntimeError(
                "WebGIS viste status 'Færdig', men "
                "loading-logoet forsvandt ikke."
            ) from exc

async def aabn_rapport(
    page: Page,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Page:
    """
    Klikker Rapport og Vis rapport.

    Output:
        Returnerer den nye Page, som viser PDF-rapporten.
        Hvis WebGIS mod forventning genbruger samme fane, returneres
        den eksisterende Page.
    """

    rapport_knap = page.locator(
        RAPPORT_KNAP
    ).filter(
        has_text=re.compile(
            r"^\s*Rapport\s*$",
            re.IGNORECASE,
        )
    ).first

    await _vent_paa_synlig(
        rapport_knap,
        navn="Rapport-knappen",
        timeout_ms=timeout_ms,
    )

    logger.info(
        "Åbner WebGIS-rapportdialogen."
    )

    await rapport_knap.click()

    await _vent_mellem_selector_actions(
        page
    )

    vis_rapport_knap = page.locator(
        VIS_RAPPORT_KNAP
    ).filter(
        has_text=re.compile(
            r"^\s*Vis rapport\s*$",
            re.IGNORECASE,
        )
    ).first

    await _vent_paa_aktiv(
        vis_rapport_knap,
        navn="Vis rapport-knappen",
        timeout_ms=timeout_ms,
    )

    loading = page.locator(
        RAPPORT_LOADING
    ).first

    sider_foer = set(
        page.context.pages
    )

    logger.info(
        "Bestiller den færdige PDF-rapport."
    )

    async with page.context.expect_page(
        timeout=timeout_ms,
    ) as ny_side_info:
        await vis_rapport_knap.click()

        await _vent_mellem_selector_actions(
            page
        )

    rapport_side = await ny_side_info.value

    await rapport_side.wait_for_load_state(
        "domcontentloaded",
        timeout=timeout_ms,
    )

    if await loading.count() > 0:
        try:
            await loading.wait_for(
                state="hidden",
                timeout=timeout_ms,
            )
        except PlaywrightTimeoutError:
            logger.debug(
                "Rapport-loading kunne ikke længere findes "
                "på den oprindelige side."
            )

    if rapport_side in sider_foer:
        logger.warning(
            "WebGIS genbrugte en eksisterende rapportfane."
        )

    await _vent_paa_pdf_url(
        rapport_side=rapport_side,
        timeout_ms=timeout_ms,
    )

    logger.info(
        "PDF-rapporten er åbnet i en ny fane."
    )

    return rapport_side

async def gem_rapport_pdf(
    webgis_side: Page,
    rapport_side: Page,
    destination: str | Path,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Path:
    """
    Henter PDF-rapporten gennem den autentificerede WebGIS-side.

    Browserens fetch-funktion bruges med credentials="include".
    Dermed genbruges browserens eksisterende HTTP-login og cookies.

    Parametre:
        webgis_side:
            Den oprindelige, autentificerede WebGIS-side.

        rapport_side:
            Den nye fane, som viser PDF-rapporten.

        destination:
            Stien, hvor PDF-filen skal gemmes.

        timeout_ms:
            Maksimal ventetid for hentning af PDF'en.

    Output:
        Path til den gemte PDF-fil.
    """

    _valider_page(
        webgis_side
    )

    _valider_page(
        rapport_side
    )

    destination_path = Path(
        destination
    )

    destination_path.parent.mkdir(
        parents=True,
        exist_ok=True,
    )

    await _vent_paa_pdf_url(
        rapport_side=rapport_side,
        timeout_ms=timeout_ms,
    )

    rapport_url = str(
        rapport_side.url or ""
    ).strip()

    if not rapport_url:
        raise RuntimeError(
            "Rapportfanen mangler en URL."
        )

    logger.info(
        "Henter PDF-rapporten gennem den "
        "autentificerede WebGIS-side."
    )

    browser_resultat = await webgis_side.evaluate(
        """
        async ({
            rapportUrl,
            timeoutMs
        }) => {
            const controller = new AbortController();

            const timeoutHandle = setTimeout(
                () => controller.abort(),
                timeoutMs
            );

            try {
                const response = await fetch(
                    rapportUrl,
                    {
                        method: "GET",
                        credentials: "include",
                        headers: {
                            "Accept": "application/pdf"
                        },
                        signal: controller.signal
                    }
                );

                const contentType = (
                    response.headers.get(
                        "content-type"
                    ) || ""
                );

                if (!response.ok) {
                    return {
                        ok: false,
                        status: response.status,
                        statusText: response.statusText,
                        contentType: contentType,
                        base64: "",
                        browserError: ""
                    };
                }

                const arrayBuffer =
                    await response.arrayBuffer();

                const bytes =
                    new Uint8Array(arrayBuffer);

                const chunkSize = 32_768;

                let binaryText = "";

                for (
                    let index = 0;
                    index < bytes.length;
                    index += chunkSize
                ) {
                    const chunk = bytes.subarray(
                        index,
                        Math.min(
                            index + chunkSize,
                            bytes.length
                        )
                    );

                    binaryText += (
                        String.fromCharCode(...chunk)
                    );
                }

                return {
                    ok: true,
                    status: response.status,
                    statusText: response.statusText,
                    contentType: contentType,
                    base64: btoa(binaryText),
                    byteLength: bytes.length,
                    browserError: ""
                };
            } catch (error) {
                return {
                    ok: false,
                    status: 0,
                    statusText: "",
                    contentType: "",
                    base64: "",
                    byteLength: 0,
                    browserError: (
                        error instanceof Error
                            ? (
                                error.name
                                + ": "
                                + error.message
                            )
                            : String(error)
                    )
                };
            } finally {
                clearTimeout(
                    timeoutHandle
                );
            }
        }
        """,
        {
            "rapportUrl": rapport_url,
            "timeoutMs": timeout_ms,
        },
    )

    if not isinstance(
        browser_resultat,
        dict,
    ):
        raise RuntimeError(
            "WebGIS returnerede et ugyldigt resultat "
            "ved hentning af PDF-rapporten."
        )

    browserfejl = str(
        browser_resultat.get(
            "browserError"
        )
        or ""
    ).strip()

    if browserfejl:
        raise RuntimeError(
            "PDF-rapporten kunne ikke hentes gennem "
            "WebGIS-browseren. "
            f"Browserfejl: {browserfejl}"
        )

    status = browser_resultat.get(
        "status"
    )

    if not browser_resultat.get(
        "ok"
    ):
        raise RuntimeError(
            "PDF-rapporten kunne ikke hentes gennem "
            "WebGIS-browseren. "
            f"HTTP-status: {status}. "
            f"Statusbesked: "
            f"{browser_resultat.get('statusText')!s}. "
            f"URL: {_sikker_url(rapport_url)}"
        )

    base64_pdf = str(
        browser_resultat.get(
            "base64"
        )
        or ""
    ).strip()

    if not base64_pdf:
        raise RuntimeError(
            "WebGIS returnerede et tomt PDF-resultat."
        )

    try:
        import base64

        pdf_bytes = base64.b64decode(
            base64_pdf,
            validate=True,
        )
    except Exception as exc:
        raise RuntimeError(
            "PDF-resultatet fra WebGIS kunne ikke "
            "Base64-dekodes."
        ) from exc

    if not pdf_bytes.startswith(
        b"%PDF-"
    ):
        content_type = str(
            browser_resultat.get(
                "contentType"
            )
            or ""
        ).strip()

        raise RuntimeError(
            "WebGIS returnerede ikke en gyldig PDF. "
            f"Content-Type var {content_type!r}, "
            "og filen manglede PDF-signaturen."
        )

    destination_path.write_bytes(
        pdf_bytes
    )

    gemt_filstoerrelse = (
        destination_path.stat().st_size
    )

    if gemt_filstoerrelse == 0:
        raise RuntimeError(
            "Den gemte PDF-rapport er tom."
        )

    forventet_filstoerrelse = (
        browser_resultat.get(
            "byteLength"
        )
    )

    if (
        isinstance(
            forventet_filstoerrelse,
            int,
        )
        and forventet_filstoerrelse > 0
        and gemt_filstoerrelse
        != forventet_filstoerrelse
    ):
        raise RuntimeError(
            "Den gemte PDF-fil har ikke den "
            "forventede størrelse. "
            f"Forventede {forventet_filstoerrelse} bytes, "
            f"men gemte {gemt_filstoerrelse} bytes."
        )

    logger.info(
        "PDF-rapporten er gemt med %s bytes.",
        gemt_filstoerrelse,
    )

    return destination_path

async def _vent_paa_synlig(

    *,
    navn: str,
    timeout_ms: int,

    """
    Venter på, at et element bliver synligt.
    """

        await locator.wait_for(
            state="visible",
            timeout=timeout_ms,
        )
    except PlaywrightTimeoutError as exc:
        raise RuntimeError(
            f"WebGIS-elementet {navn!r} blev ikke synligt."
        ) from exc


async def _vent_paa_aktiv(
    locator: Locator,
    *,
    navn: str,
    timeout_ms: int,
) -> None:
    """
    Venter på, at et element er synligt og aktiveret.
    """

    await _vent_paa_synlig(
        locator,
        navn=navn,
        timeout_ms=timeout_ms,
    )

    try:
        await locator.wait_for(
            state="attached",
            timeout=timeout_ms,
        )

        await locator.evaluate(
            """
            (element, timeoutMs) => new Promise(
                (resolve, reject) => {
                    const startedAt = Date.now();

                    const check = () => {
                        const disabled = (
                            element.disabled === true
                            || element.getAttribute(
                                "aria-disabled"
                            ) === "true"
                        );

                        if (!disabled) {
                            resolve(true);
                            return;
                        }

                        if (
                            Date.now() - startedAt
                            >= timeoutMs
                        ) {
                            reject(
                                new Error(
                                    "Elementet forblev disabled."
                                )
                            );
                            return;
                        }

                        setTimeout(check, 100);
                    };

                    check();
                }
            )
            """,
            timeout_ms,
        )
    except Exception as exc:
        raise RuntimeError(
            f"WebGIS-elementet {navn!r} blev ikke aktivt."
        ) from exc

def _valider_wkt(
    wkt: str,
) -> str:
    """
    Kontrollerer en grundlæggende WKT-værdi.

    Output:
        Trimmet WKT.
    """

    if not isinstance(wkt, str):
        raise TypeError(
            "wkt skal være tekst."
        )

    valideret_wkt = wkt.strip()

    if not valideret_wkt:
        raise ValueError(
            "wkt må ikke være tom."
        )

    tilladte_geometrityper = (
        "POINT",
        "LINESTRING",
        "POLYGON",
        "MULTIPOINT",
        "MULTILINESTRING",
        "MULTIPOLYGON",
        "GEOMETRYCOLLECTION",
    )

    wkt_upper = valideret_wkt.upper()

    if not wkt_upper.startswith(
        tilladte_geometrityper
    ):
        raise ValueError(
            "wkt starter ikke med en understøttet "
            "WKT-geometritype."
        )

    if (
        valideret_wkt.count("(")
        != valideret_wkt.count(")")
    ):
        raise ValueError(
            "wkt har ikke samme antal åbne og lukkede parenteser."
        )

    return valideret_wkt


def _valider_tema(
    tema: str,
) -> str:
    """
    Kontrollerer dropdown-værdien.

    Output:
        Trimmet temaværdi.
    """

    if not isinstance(tema, str):
        raise TypeError(
            "tema skal være tekst."
        )

    valideret_tema = tema.strip()

    if valideret_tema not in WEBGIS_TILLADTE_TEMAER:
        raise ValueError(
            f"Ukendt WebGIS-tema: {tema!r}. "
            f"Tilladte værdier: {WEBGIS_TILLADTE_TEMAER}."
        )

    return valideret_tema


def _valider_pdf_filnavn(
    filnavn: str,
) -> str:
    """
    Kontrollerer rapportens filnavn.

    Output:
        Et sikkert filnavn, som ender på .pdf.
    """

    if not isinstance(filnavn, str):
        raise TypeError(
            "filnavn skal være tekst."
        )

    valideret_filnavn = filnavn.strip()

    if not valideret_filnavn:
        raise ValueError(
            "filnavn må ikke være tomt."
        )

    if Path(valideret_filnavn).name != valideret_filnavn:
        raise ValueError(
            "filnavn må ikke indeholde en mappesti."
        )

    if not valideret_filnavn.casefold().endswith(
        ".pdf"
    ):
        valideret_filnavn += ".pdf"

    return valideret_filnavn


def _opret_rapportsti(
    *,
    rapportmappe: str | Path,
    filnavn: str,
) -> Path:
    """
    Opretter rapportmappen.

    Output:
        Den komplette sti til PDF-filen.
    """

    rapportmappe_path = Path(
        rapportmappe
    )

    rapportmappe_path.mkdir(
        parents=True,
        exist_ok=True,
    )

    return (
        rapportmappe_path
        / filnavn
    )


def _sikker_url(
    url: str,
) -> str:
    """
    Returnerer URL uden query-parametre eller credentials.
    """

    parsed_url = urlparse(
        str(url or "")
    )

    return parsed_url._replace(
        query="",
        fragment="",
        params="",
    ).geturl()

from __future__ import annotations

import logging
from typing import Any

from playwright.async_api import Page


logger = logging.getLogger(__name__)


# ------------------------------------------------------------

SPATIALANALYZE_URL = (
    "https://webgis.intern.udcit.dk/"
    "spatialmap?page=spatialanalyze"
)

DEFAULT_TIMEOUT_MS = 60_000

# ------------------------------------------------------------


async def spatialanalyze(
    page: Page,
    session_id: str,
    datasource: str,
    min_x_wc: float,
    min_y_wc: float,
    max_x_wc: float,
    max_y_wc: float,
    maxrows: int,
    *,
    command: str,
    spatialexpression: str,
    jsontypes: str,
    outputformat: str,
    error_content_type: str,
    profile: str = "maxi",
    timeout_ms: int = DEFAULT_TIMEOUT_MS,
) -> dict[str, Any]:
    """Udfører WebGIS spatialanalyze i den åbne browser.

    Kaldet sendes som application/x-www-form-urlencoded gennem
    browserens egen fetch-funktion. Derfor genbruges browserens
    eksisterende HTTP-login og JSESSIONID.

    Parametre:
        page:
            Den allerede åbnede og autentificerede WebGIS-side.

        session_id:
            Session-Id opsnappet fra WebGIS' egne requests.

        datasource:
            WebGIS-datasourcen, eksempelvis
            "ds-byggesag_cases_matrikel".

        min_x_wc, min_y_wc, max_x_wc, max_y_wc:
            Koordinaterne til analyseområdet.

        maxrows:
            Maksimalt antal resultatrækker.

        command:
            Den præcise command-værdi fra Request 21.

        spatialexpression:
            Den præcise spatialexpression-værdi fra Request 21.

        jsontypes:
            Den præcise jsontypes-værdi fra Request 21.

        outputformat:
            Den præcise outputformat-værdi fra Request 21.

        error_content_type:
            Den præcise værdi for jdaf.error.contenttype
            fra Request 21.

        profile:
            WebGIS-profil. Standarden er "maxi".

        timeout_ms:
            Maksimal ventetid for browserkaldet.

    Output:
        WebGIS-svaret som en dictionary.

    Sikkerhed:
        Session-Id, cookies og formularindhold logges ikke.
    """
    _valider_page(page)

    session_id = _valider_tekst(
        value=session_id,
        field_name="session_id",
    )

    datasource = _valider_tekst(
        value=datasource,
        field_name="datasource",
    )

    command = _valider_tekst(
        value=command,
        field_name="command",
    )

    spatialexpression = _valider_tekst(
        value=spatialexpression,
        field_name="spatialexpression",
    )

    jsontypes = _valider_tekst(
        value=jsontypes,
        field_name="jsontypes",
    )

    outputformat = _valider_tekst(
        value=outputformat,
        field_name="outputformat",
    )

    error_content_type = _valider_tekst(
        value=error_content_type,
        field_name="error_content_type",
    )

    profile = _valider_tekst(
        value=profile,
        field_name="profile",
    )

    maxrows = _valider_positivt_heltal(
        value=maxrows,
        field_name="maxrows",
    )

    timeout_ms = _valider_positivt_heltal(
        value=timeout_ms,
        field_name="timeout_ms",
    )

    koordinater = _valider_koordinater(
        min_x_wc=min_x_wc,
        min_y_wc=min_y_wc,
        max_x_wc=max_x_wc,
        max_y_wc=max_y_wc,
    )

    formular_data = {
        "page": "spatialanalyze",
        "profile": profile,
        "command": command,
        "datasource": datasource,
        "spatialexpression": spatialexpression,
        "min_x_wc": str(koordinater["min_x_wc"]),
        "min_y_wc": str(koordinater["min_y_wc"]),
        "max_x_wc": str(koordinater["max_x_wc"]),
        "max_y_wc": str(koordinater["max_y_wc"]),
        "maxrows": str(maxrows),
        "outputformat": outputformat,
        "jsontypes": jsontypes,
        "jdaf.error.contenttype": error_content_type,
    }

    logger.info(
        "Starter WebGIS spatialanalyze for datasource=%s, "
        "maxrows=%s.",
        datasource,
        maxrows,
    )

    browser_resultat = await page.evaluate(
        """
        async ({
            url,
            sessionId,
            formValues,
            timeoutMs
        }) => {
            const controller = new AbortController();

            const timeoutHandle = setTimeout(
                () => controller.abort(),
                timeoutMs
            );

            try {
                const formBody = new URLSearchParams();

                for (
                    const [fieldName, fieldValue]
                    of Object.entries(formValues)
                ) {
                    formBody.append(
                        fieldName,
                        String(fieldValue)
                    );
                }

                const response = await fetch(
                    url,
                    {
                        method: "POST",
                        credentials: "include",
                        headers: {
                            "Accept": (
                                "application/json, "
                                + "text/javascript, "
                                + "*/*; q=0.01"
                            ),
                            "Content-Type": (
                                "application/"
                                + "x-www-form-urlencoded; "
                                + "charset=UTF-8"
                            ),
                            "Session-Id": sessionId,
                            "X-Requested-With": (
                                "XMLHttpRequest"
                            )
                        },
                        body: formBody.toString(),
                        signal: controller.signal
                    }
                );

                const responseText =
                    await response.text();

                return {
                    ok: response.ok,
                    status: response.status,
                    statusText: response.statusText,
                    contentType: (
                        response.headers.get(
                            "content-type"
                        ) || ""
                    ),
                    responseText: responseText,
                    finalUrl: response.url
                };
            } catch (error) {
                return {
                    ok: false,
                    status: 0,
                    statusText: "",
                    contentType: "",
                    responseText: "",
                    finalUrl: "",
                    browserError: (
                        error instanceof Error
                            ? (
                                error.name
                                + ": "
                                + error.message
                            )
                            : String(error)
                    )
                };
            } finally {
                clearTimeout(timeoutHandle);
            }
        }
        """,
        {
            "url": SPATIALANALYZE_URL,
            "sessionId": session_id,
            "formValues": formular_data,
            "timeoutMs": timeout_ms,
        },
    )

    fetch_resultat = _valider_fetch_resultat(
        browser_resultat
    )

    status = fetch_resultat["status"]

    browser_error = str(
        fetch_resultat.get("browserError")
        or ""
    ).strip()

    if browser_error:
        raise RuntimeError(
            "WebGIS spatialanalyze kunne ikke udføres "
            "i browseren. "
            f"Browserfejl: {browser_error}"
        )

    if not fetch_resultat["ok"]:
        response_preview = _lav_sikkert_response_preview(
            fetch_resultat.get("responseText")
        )

        raise RuntimeError(
            "WebGIS spatialanalyze fejlede med "
            f"HTTP-status {status}. "
            f"Statusbesked: "
            f"{fetch_resultat.get('statusText')!s}. "
            f"Svareksempel: {response_preview}"
        )

    response_text = str(
        fetch_resultat.get("responseText")
        or ""
    ).strip()

    if not response_text:
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede "
            "et tomt svar."
        )

    resultat = await _parse_json_i_browser(
        page=page,
        response_text=response_text,
    )

    if not isinstance(resultat, dict):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede JSON, "
            "men resultatet var ikke en dictionary. "
            f"Modtog: {type(resultat).__name__}."
        )

    logger.info(
        "WebGIS spatialanalyze blev gennemført med "
        "HTTP-status %s.",
        status,
    )

    return resultat

async def _parse_json_i_browser(
    page: Page,
    response_text: str,
) -> Any:
    """Parser WebGIS-svaret som JSON.

    Output:
        Den parsede JSON-værdi.
    """
    parse_resultat = await page.evaluate(
        """
        (responseText) => {
            try {
                return {
                    parsed: true,
                    value: JSON.parse(responseText)
                };
            } catch (error) {
                return {
                    parsed: false,
                    error: (
                        error instanceof Error
                            ? error.message
                            : String(error)
                    )
                };
            }
        }
        """,
        response_text,
    )

    if not isinstance(parse_resultat, dict):
        raise RuntimeError(
            "WebGIS-svaret kunne ikke kontrolleres "
            "som JSON."
        )

    if not parse_resultat.get("parsed"):
        response_preview = (
            _lav_sikkert_response_preview(
                response_text
            )
        )

        raise RuntimeError(
            "WebGIS spatialanalyze returnerede ikke "
            "gyldig JSON. "
            f"JSON-fejl: "
            f"{parse_resultat.get('error')!s}. "
            f"Svareksempel: {response_preview}"
        )

    return parse_resultat.get("value")

def _valider_fetch_resultat(
    value: Any,
) -> dict[str, Any]:
    """Validerer resultatet fra browserens fetch-kald."""
    if not isinstance(value, dict):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede et "
            "ugyldigt browserresultat."
        )

    status = value.get("status")

    if isinstance(status, bool):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede en "
            "ugyldig HTTP-status."
        )

    if not isinstance(status, int):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede ikke "
            "en numerisk HTTP-status."
        )

    if not isinstance(value.get("ok"), bool):
        raise RuntimeError(
            "WebGIS spatialanalyze returnerede ikke "
            "en gyldig ok-status."
        )

    return value

def _valider_positivt_heltal(

) -> int:

    except (TypeError, ValueError) as exc:

        ) from exc

    if integer_value <= 0:
        raise ValueError(
            f"{field_name} skal være større end 0."
        )

    return integer_value

    """Validerer spatialanalysens koordinater."""
    koordinater = {

            field_name="min_x_wc",

        "min_y_wc": _valider_koordinat(

    }

    if (
        koordinater["min_x_wc"]
        >= koordinater["max_x_wc"]
    ):
        raise ValueError(
            "min_x_wc skal være mindre end max_x_wc."
        )

    if (
        koordinater["min_y_wc"]
        >= koordinater["max_y_wc"]
    ):
        raise ValueError(
            "min_y_wc skal være mindre end max_y_wc."
        )

    return koordinater

def _lav_sikkert_response_preview(
    value: Any,
    maximum_length: int = 500,
) -> str:
    """Laver et begrænset preview af et fejlsvar.

    Previewet bruges kun til fejlsøgning. Session-Id og cookies
    ligger ikke i response-bodyen og bliver ikke tilføjet her.
    """
    text = " ".join(
        str(value or "").split()
    )

    if not text:
        return "<tomt svar>"

    if len(text) <= maximum_length:
        return text

    return (
        text[:maximum_length]
        + "..."
    )

        "page": "spatialanalyze-get-stored-data",

        "Henter op til %s rækker fra den seneste "

        """
        async ({
            url,
            sessionId,
            formValues,
            timeoutMs
        }) => {
            const controller = new AbortController();

            const timeoutHandle = setTimeout(
                () => controller.abort(),
                timeoutMs
            );

            try {
                const formBody = new URLSearchParams();

                for (
                    const [fieldName, fieldValue]
                    of Object.entries(formValues)
                ) {
                    formBody.append(
                        fieldName,
                        String(fieldValue)
                    );
                }

                const response = await fetch(
                    url,
                    {
                        method: "POST",
                        credentials: "include",
                        headers: {
                            "Accept": (
                                "application/json, "
                                + "text/javascript, "
                                + "*/*; q=0.01"
                            ),
                            "Content-Type": (
                                "application/"
                                + "x-www-form-urlencoded; "
                                + "charset=UTF-8"
                            ),
                            "Session-Id": sessionId,
                            "X-Requested-With": "XMLHttpRequest"
                        },
                        body: formBody.toString(),
                        signal: controller.signal
                    }
                );

                const responseText =
                    await response.text();

                return {
                    ok: response.ok,
                    status: response.status,
                    statusText: response.statusText,
                    contentType: (
                        response.headers.get(
                            "content-type"
                        ) || ""
                    ),
                    responseText: responseText
                };
            } catch (error) {
                return {
                    ok: false,
                    status: 0,
                    statusText: "",
                    contentType: "",
                    responseText: "",
                    browserError: (
                        error instanceof Error
                            ? (
                                error.name
                                + ": "
                                + error.message
                            )
                            : String(error)
                    )
                };
            } finally {
                clearTimeout(timeoutHandle);
            }
        }
        """,
        {
            "url": SPATIALANALYZE_STORED_DATA_URL,
            "sessionId": session_id,
            "formValues": formular_data,
            "timeoutMs": timeout_ms,

        browser_resultat
    )

        fetch_resultat.get("browserError")

            f"Browserfejl: {browser_error}"

                )

def _find_datarækker(
    value: Any,
) -> list[dict[str, Any]]:
    """Finder den mest sandsynlige liste med datarækker.

    WebGIS kan pakke rækkerne ind i forskellige JSON-strukturer.
    Funktionen gennemgår derfor dictionaries og lister rekursivt.

    Output:
        Den største liste med dictionaries, som ligner datarækker.
    """
    kandidater: list[
        list[dict[str, Any]]
    ] = []

    def besoeg(
        current_value: Any,
    ) -> None:
        if isinstance(
            current_value,
            list,
        ):
            dictionary_rows = [
                element
                for element in current_value
                if isinstance(element, dict)
                and not _er_metadata_raekke(element)
            ]

            if dictionary_rows:
                kandidater.append(
                    dictionary_rows
                )

            for element in current_value:
                besoeg(element)

            return

        if isinstance(
            current_value,
            dict,
        ):
            for nested_value in (
                current_value.values()
            ):
                besoeg(nested_value)

    besoeg(value)

    if not kandidater:
        if (
            isinstance(value, dict)
            and not _er_metadata_raekke(value)
        ):
            return [value]

        return []

    return max(
        kandidater,
        key=len,
    )


def _er_metadata_raekke(
    row: dict[str, Any],
) -> bool:
    """Kontrollerer om en række kun ligner WebGIS-metadata."""
    if not isinstance(row, dict):
        return True

    keys = {
        str(key).casefold()
        for key in row
    }

    metadata_keys = {
        "_name",
        "_class",
        "_pcolid",
        "rowcount",
        "xmin",
        "ymin",
        "xmax",
        "ymax",
    }

    return bool(keys) and keys.issubset(
        metadata_keys
    )

async def vent_paa_jsessionid(
    page: Page,
    timeout_seconds: int = COOKIE_TIMEOUT_SECONDS,
) -> bool:
    """Venter på JSESSIONID i den aktuelle browserkontekst.

    Output:
        True, når JSESSIONID findes.

    Sikkerhed:
        Cookieværdien bliver aldrig udskrevet eller returneret.
    """
    antal_forsog = timeout_seconds * 2

    for _ in range(antal_forsog):
        cookies = await page.context.cookies(
            [WEBGIS_URL]
        )

        har_jsessionid = any(
            cookie.get("name", "").casefold()
            == JSESSIONID_COOKIE_NAME.casefold()
            and bool(cookie.get("value"))
            for cookie in cookies
        )

        if har_jsessionid:
            logger.info(
                "WebGIS-sessionens JSESSIONID er fundet."
            )
            return True

        await asyncio.sleep(0.5)

    raise RuntimeError(
        "WebGIS blev åbnet, men JSESSIONID blev ikke "
        f"fundet inden for {timeout_seconds} sekunder."
    )


async def udskriv_cookie_navne(

        {
            str(cookie.get("name", "")).strip()
            for cookie in cookies
            if cookie.get("name")
        }
    )

    logger.info(
        "Cookienavne efter WebGIS-login: %s",
        cookie_names,
    )

def opret_session_id_listener(
    page: Page,
) -> asyncio.Future:
    event_loop = asyncio.get_running_loop()
    session_id_future: asyncio.Future[str] = event_loop.create_future()

    async def kontroller_request(request: Request) -> None:
        if session_id_future.done():
            return

        request_hostname = (urlparse(request.url).hostname or "").casefold()

        if request_hostname != WEBGIS_HOSTNAME:
            return

        try:
            headers = await request.all_headers()
        except Exception as exc:
            logger.debug(
                "Headers kunne ikke læses fra et WebGIS-request: %s",
                exc,
            )
            return

        session_id = str(
            headers.get(SESSION_ID_HEADER_NAME, "") or ""
        ).strip()

        if not session_id or session_id_future.done():
            return

        logger.info("WebGIS Session-Id er fundet automatisk.")
        session_id_future.set_result(session_id)

    def log_listenerfejl(completed_task: asyncio.Task[None]) -> None:
        if completed_task.cancelled():
            return

        try:
            exception = completed_task.exception()
        except asyncio.CancelledError:
            return

        if exception is not None:
            logger.error(
                "Fejl i Session-Id-listeneren: %s",
                exception,
                exc_info=(
                    type(exception),
                    exception,
                    exception.__traceback__,
                ),
            )

    def ved_request(request: Request) -> None:
        task = asyncio.create_task(kontroller_request(request))
        task.add_done_callback(log_listenerfejl)

    page.on("request", ved_request)
    return session_id_future


async def vent_paa_session_id(
    session_id_future: asyncio.Future[str],
    timeout_seconds: int = SESSION_TIMEOUT_SECONDS,
) -> str:
    """Venter på Session-Id fra WebGIS' egne requests."""
    logger.info("Venter på WebGIS Session-Id.")

    try:
        session_id = await asyncio.wait_for(
            asyncio.shield(session_id_future),
            timeout=timeout_seconds,
        )
    except TimeoutError as exc:
        raise RuntimeError(
            "WebGIS blev åbnet, men Session-Id blev ikke fundet "
            f"inden for {timeout_seconds} sekunder."
        ) from exc

    session_id = str(session_id or "").strip()

    if not session_id:
        raise RuntimeError(
            "WebGIS-listeneren returnerede en tom Session-Id."
        )

    return session_id

async def launch_webgis(page: Page) -> Page:
    """Åbner WebGIS automatisk i den eksisterende browsersession."""
    logger.info("Åbner WebGIS automatisk.")

    response = await page.goto(
        WEBGIS_URL,

    if response is None:
        raise RuntimeError(
            "Navigationen til WebGIS returnerede ikke et HTTP-svar."

    if not response.ok:
        raise RuntimeError(

            f"HTTP-status {response.status}."
        )

    logger.info("WebGIS er åbnet.")
    return page

async def vent_paa_jsessionid(
    page: Page,
    timeout_seconds: int = SESSION_TIMEOUT_SECONDS,
) -> None:
    """Venter på JSESSIONID uden at returnere cookieindholdet."""
    logger.info("Venter på WebGIS-sessionens JSESSIONID.")

    event_loop = asyncio.get_running_loop()
    deadline = event_loop.time() + timeout_seconds

    while event_loop.time() < deadline:
        cookies = await page.context.cookies([WEBGIS_BASE_URL])

        har_jsessionid = any(
            str(cookie.get("name", "")).casefold()
            == JSESSIONID_COOKIE_NAME.casefold()
            and bool(cookie.get("value"))
            for cookie in cookies
        )

        if har_jsessionid:
            logger.info("WebGIS-sessionens JSESSIONID er fundet.")
            return

        await asyncio.sleep(0.5)

    raise RuntimeError(
        "WebGIS blev åbnet, men JSESSIONID blev ikke fundet "
        f"inden for {timeout_seconds} sekunder."
    )

def valider_analyseresultat(resultat: dict[str, Any]) -> None:
    """Kontrollerer spatialanalyze-resultatets grundstruktur."""
    if not isinstance(resultat, dict):
        raise TypeError(
            "spatialanalyze() skal returnere en dictionary."
        )

    if not resultat:
        raise AssertionError(
            "spatialanalyze() returnerede en tom dictionary."
        )


def valider_raekker(raekker: list[dict[str, Any]]) -> None:
    """Kontrollerer de konkrete spatialanalyse-rækker."""
    if not isinstance(raekker, list):
        raise TypeError(
            "hent_spatialanalyze_raekker() skal returnere en liste."
        )

    for index, raekke in enumerate(raekker, start=1):
        if not isinstance(raekke, dict):
            raise TypeError(
                "Alle spatialanalyse-rækker skal være dictionaries. "
                f"Række {index} er {type(raekke).__name__}."
            )


def find_rowcount(value: Any) -> int | None:
    """Finder rowcount rekursivt i analysesvaret."""
    if isinstance(value, dict):
        for key, nested_value in value.items():
            if str(key).casefold() == "rowcount":
                try:
                    return int(nested_value)
                except (TypeError, ValueError) as exc:
                    raise ValueError(
                        "Analysens rowcount kunne ikke konverteres "
                        "til et heltal."
                    ) from exc

        for nested_value in value.values():
            found_rowcount = find_rowcount(nested_value)
            if found_rowcount is not None:
                return found_rowcount

    elif isinstance(value, list):
        for nested_value in value:
            found_rowcount = find_rowcount(nested_value)
            if found_rowcount is not None:
                return found_rowcount

    return None


def valider_antal_raekker(
    resultat: dict[str, Any],
    raekker: list[dict[str, Any]],
) -> int | None:
    """Sammenligner analysens rowcount med de hentede rækker."""
    rowcount = find_rowcount(resultat)

    if rowcount is None:
        logger.warning(
            "Analysesvaret indeholder ikke et læsbart rowcount-felt."
        )
        return None

    if rowcount < 0:
        raise AssertionError(
            f"Analysens rowcount må ikke være negativ: {rowcount}."
        )

    if rowcount > MAX_ROWS:
        raise AssertionError(
            "Analysens rowcount er større end MAX_ROWS. "
            f"Rowcount: {rowcount}. MAX_ROWS: {MAX_ROWS}."
        )

    if len(raekker) != rowcount:
        raise AssertionError(
            "Antallet af hentede rækker svarer ikke til analysens "
            f"rowcount. Rowcount: {rowcount}. "
            f"Hentede rækker: {len(raekker)}."
        )

    logger.info(
        "Antal hentede rækker matcher rowcount: %s.",
        rowcount,
    )
    return rowcount

    print("=" * 80)

_DOWNLOADS_BY_PAGE: dict[int, Download] = {}


async def _aabn_rapport_med_download(
    page: Page,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Page:
    """Klikker Vis rapport og fanger den faktiske Playwright-download."""
    rapport_knap = page.locator(RAPPORT_KNAP).filter(
        has_text=re.compile(r"^\s*Rapport\s*$", re.IGNORECASE)
    ).first
    await rapport_knap.wait_for(state="visible", timeout=timeout_ms)
    await rapport_knap.click()
    await page.wait_for_timeout(1_000)

    vis_rapport_knap = page.locator(VIS_RAPPORT_KNAP).filter(
        has_text=re.compile(r"^\s*Vis rapport\s*$", re.IGNORECASE)
    ).first
    await vis_rapport_knap.wait_for(state="visible", timeout=timeout_ms)

    loop = asyncio.get_running_loop()
    download_future: asyncio.Future[Download] = loop.create_future()

    def ved_download(download: Download) -> None:
        if not download_future.done():
            download_future.set_result(download)

    def ved_ny_side(ny_side: Page) -> None:
        ny_side.on("download", ved_download)

    page.on("download", ved_download)
    page.context.on("page", ved_ny_side)

    logger.info("Bestiller rapporten og venter på download-event.")

    try:
        async with page.context.expect_page(timeout=timeout_ms) as popup_info:
            await vis_rapport_knap.click()

        rapport_side = await popup_info.value
        rapport_side.on("download", ved_download)

        download = await asyncio.wait_for(
            asyncio.shield(download_future),
            timeout=timeout_ms / 1_000,
        )

        fejl = await download.failure()
        if fejl:
            raise RuntimeError(f"WebGIS-download fejlede: {fejl}")

        _DOWNLOADS_BY_PAGE[id(rapport_side)] = download

        logger.info(
            "Download-event modtaget. Foreslået filnavn: %s",
            download.suggested_filename,
        )
        return rapport_side
    finally:
        page.remove_listener("download", ved_download)
        page.context.remove_listener("page", ved_ny_side)


async def _gem_downloadet_rapport(
    webgis_side: Page,
    rapport_side: Page,
    destination: str | Path,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Path:
    """Gemmer den allerede modtagne download uden et nyt HTTP-kald."""
    del webgis_side, timeout_ms

    download = _DOWNLOADS_BY_PAGE.pop(id(rapport_side), None)
    if download is None:
        raise RuntimeError("Der blev ikke registreret en rapport-download.")

    destination_path = Path(destination)
    destination_path.parent.mkdir(parents=True, exist_ok=True)
    destination_path.unlink(missing_ok=True)

    await download.save_as(str(destination_path))

    if not destination_path.is_file():
        raise RuntimeError("Den downloadede PDF blev ikke gemt.")
    if destination_path.stat().st_size == 0:
        raise RuntimeError("Den downloadede PDF er tom.")
    if not destination_path.read_bytes().startswith(b"%PDF-"):
        raise RuntimeError("Den downloadede fil er ikke en gyldig PDF.")

    logger.info(
        "PDF gemt direkte fra download-event: %s bytes.",
        destination_path.stat().st_size,
    )
    return destination_path

async def _aabn_rapport_headless(
    page: Page,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Page:
    """Åbner rapporten og opsnapper PDF-responsens URL fra netværket."""
    rapport_knap = page.locator(RAPPORT_KNAP).filter(
        has_text=re.compile(r"^\s*Rapport\s*$", re.IGNORECASE)
    ).first
    await rapport_knap.wait_for(state="visible", timeout=timeout_ms)

    logger.info("Åbner WebGIS-rapportdialogen.")
    await rapport_knap.click()
    await page.wait_for_timeout(1_000)

    vis_rapport_knap = page.locator(VIS_RAPPORT_KNAP).filter(
        has_text=re.compile(r"^\s*Vis rapport\s*$", re.IGNORECASE)
    ).first
    await vis_rapport_knap.wait_for(state="visible", timeout=timeout_ms)

    pdf_url_future: asyncio.Future[str] = (
        asyncio.get_running_loop().create_future()
    )

    def ved_response(response: Response) -> None:
        if pdf_url_future.done():
            return

        content_type = str(
            response.headers.get("content-type", "")
        ).casefold()
        content_disposition = str(
            response.headers.get("content-disposition", "")
        ).casefold()
        response_url = str(response.url or "").strip()

        er_pdf = (
            "application/pdf" in content_type
            or ".pdf" in response_url.casefold()
            or "filename=" in content_disposition
            and ".pdf" in content_disposition
        )

        if er_pdf and response_url:
            logger.info(
                "PDF-respons opsnappet: content-type=%s",
                content_type or "ukendt",
            )
            pdf_url_future.set_result(response_url)

    page.context.on("response", ved_response)

    logger.info("Bestiller den færdige PDF-rapport.")

    try:
        async with page.context.expect_page(
            timeout=timeout_ms,
        ) as ny_side_info:
            await vis_rapport_knap.click()

        rapport_side = await ny_side_info.value

        pdf_url = await asyncio.wait_for(
            asyncio.shield(pdf_url_future),
            timeout=timeout_ms / 1_000,
        )

        _PDF_URL_BY_PAGE[id(rapport_side)] = pdf_url
        logger.info("PDF-URL gemt fra netværksresponsen.")

        loading = page.locator(RAPPORT_LOADING).first
        if await loading.count() > 0:
            try:
                await loading.wait_for(
                    state="hidden",
                    timeout=timeout_ms,
                )
            except Exception:
                logger.debug(
                    "Rapport-loading kunne ikke længere findes."
                )

        return rapport_side
    finally:
        page.context.remove_listener("response", ved_response)


async def _gem_rapport_pdf_headless(
    webgis_side: Page,
    rapport_side: Page,
    destination: str | Path,
    *,
    timeout_ms: int = WEBGIS_RAPPORT_TIMEOUT_MS,
) -> Path:
    """Henter PDF'en direkte via browserkontekstens autentificerede request."""
    pdf_url = _PDF_URL_BY_PAGE.get(id(rapport_side), "").strip()
    if not pdf_url:
        raise RuntimeError(
            "Testen opsnappede ikke en PDF-URL for rapportfanen."
        )

    destination_path = Path(destination)
    destination_path.parent.mkdir(parents=True, exist_ok=True)

    logger.info(
        "Henter PDF-responsen gennem BrowserContexts request-klient."
    )
    response = await webgis_side.context.request.get(
        pdf_url,
        headers={"Accept": "application/pdf"},
        timeout=timeout_ms,
    )

    if not response.ok:
        response_text = await response.text()
        raise RuntimeError(
            "PDF-kaldet fejlede. "
            f"HTTP-status: {response.status}. "
            f"Svar: {response_text[:500]}"
        )

    pdf_bytes = await response.body()
    content_type = str(
        response.headers.get("content-type", "")
    ).strip()

    if not pdf_bytes.startswith(b"%PDF-"):
        raise RuntimeError(
            "WebGIS returnerede ikke en gyldig PDF. "
            f"Content-Type: {content_type!r}."
        )

    destination_path.write_bytes(pdf_bytes)

    if destination_path.stat().st_size == 0:
        raise RuntimeError("Den gemte PDF er tom.")

    logger.info(
        "PDF gemt: %s bytes.",
        destination_path.stat().st_size,
    )
    return destination_path


def _hent_credentials() -> dict[str, str]:
    AutomationServer.from_environment()
    credential = Credential.get_credential(WEBGIS_CREDENTIAL_NAME)
    username = str(credential.username or "").strip()
    password = str(credential.password or "")

    if not username or not password:
        raise RuntimeError(
            f"Credentialen {WEBGIS_CREDENTIAL_NAME!r} er ufuldstændig."
        )

    return {"username": username, "password": password}

def parse_wkt_polygon(
    wkt: str,
) -> list[tuple[float, float]]:
    """
    Parser en enkel WKT POLYGON-værdi.

    Output:
        Liste med polygonpunkter som (x, y).
    """

    if not isinstance(wkt, str):
        raise TypeError(
            "wkt skal være tekst."
        )

    text = wkt.strip()

    if not text:
        raise ValueError(
            "wkt må ikke være tom."
        )

    start = "POLYGON (("
    slut = "))"

    if not text.upper().startswith(start):
        raise ValueError(
            "Kun en enkel WKT POLYGON understøttes."
        )

    if not text.endswith(slut):
        raise ValueError(
            "WKT-polygonen mangler afsluttende parenteser."
        )

    koordinattekst = text[
        len(start):-len(slut)
    ]

    points: list[tuple[float, float]] = []

    for indeks, pair in enumerate(
        koordinattekst.split(","),
        start=1,
    ):
        dele = pair.strip().split()

        if len(dele) != 2:
            raise ValueError(
                "WKT-koordinatparret på position "
                f"{indeks} indeholder ikke præcis x og y."
            )

        try:
            x = float(dele[0])
            y = float(dele[1])
        except ValueError as exc:
            raise ValueError(
                "WKT-koordinatparret på position "
                f"{indeks} indeholder ugyldige tal."
            ) from exc

        points.append(
            (
                x,
                y,
            )
        )

    if len(points) < 4:
        raise ValueError(
            "En polygon skal indeholde mindst fire punkter."
        )

    if points[0] != points[-1]:
        raise ValueError(
            "WKT-polygonens første og sidste punkt skal være ens."
        )

    return points


def calculate_bbox(
    points: list[tuple[float, float]],
    buffer_meter: float = 50.0,
) -> tuple[float, float, float, float]:
    """
    Beregner polygonens bounding box med buffer.

    Output:
        xmin, ymin, xmax, ymax.
    """

    if not points:
        raise ValueError(
            "points må ikke være tom."
        )

    if isinstance(buffer_meter, bool):
        raise TypeError(
            "buffer_meter skal være et tal."
        )

    try:
        buffer_vaerdi = float(
            buffer_meter
        )
    except (TypeError, ValueError) as exc:
        raise TypeError(
            "buffer_meter skal være et tal."
        ) from exc

    if buffer_vaerdi < 0:
        raise ValueError(
            "buffer_meter må ikke være negativ."
        )

    xs = [x for x, _ in points]
    ys = [y for _, y in points]

    xmin = min(xs) - buffer_vaerdi
    ymin = min(ys) - buffer_vaerdi
    xmax = max(xs) + buffer_vaerdi
    ymax = max(ys) + buffer_vaerdi

    if xmin >= xmax or ymin >= ymax:
        raise ValueError(
            "Den beregnede bounding box er ugyldig."
        )

    return (
        xmin,
        ymin,
        xmax,
        ymax,
    )


def world_to_pixel(
    x: float,
    y: float,
    bbox: tuple[float, float, float, float],
    width: int,
    height: int,
) -> tuple[int, int]:
    """
    Konverterer en koordinat til en pixelposition.

    Output:
        Pixelpositionen som (x, y).
    """

    valider_billedstoerrelse(
        width,
        height,
    )

    xmin, ymin, xmax, ymax = bbox

    if xmin >= xmax or ymin >= ymax:
        raise ValueError(
            "bbox er ugyldig."
        )

    px = (
        (x - xmin)
        / (xmax - xmin)
        * width
    )

    py = height - (
        (y - ymin)
        / (ymax - ymin)
        * height
    )

    return (
        round(px),
        round(py),
    )

def hent_kort(
    bbox: tuple[float, float, float, float],
    *,
    token: str,
    width: int = WIDTH,
    height: int = HEIGHT,
    timeout_seconds: int = REQUEST_TIMEOUT_SECONDS,
) -> Image.Image:
    """
    Henter et baggrundskort fra Dataforsyningens WMS.

    Output:
        Et færdigindlæst Pillow-billede i RGBA-format.
    """

    if not isinstance(token, str) or not token.strip():
        raise ValueError(
            "token skal være en ikke-tom tekst."
        )

    valider_billedstoerrelse(
        width,
        height,
    )

    if isinstance(timeout_seconds, bool) or not isinstance(
        timeout_seconds,
        int,
    ):
        raise TypeError(
            "timeout_seconds skal være et heltal."
        )

    if timeout_seconds <= 0:
        raise ValueError(
            "timeout_seconds skal være større end 0."
        )

    xmin, ymin, xmax, ymax = bbox

    params = {
        "SERVICE": "WMS",
        "VERSION": "1.1.1",
        "REQUEST": "GetMap",
        "FORMAT": "image/png",
        "TRANSPARENT": "TRUE",
        "LAYERS": "Basis_kort",
        "token": token.strip(),
        "servicename": "forvaltning2",
        "SRS": "EPSG:25832",
        "STYLES": "",
        "WIDTH": width,
        "HEIGHT": height,
        "BBOX": f"{xmin},{ymin},{xmax},{ymax}",
    }

    response = requests.get(
        DATAFORSYNING_WMS_URL,
        params=params,
        timeout=timeout_seconds,
    )

    response.raise_for_status()

    content_type = str(
        response.headers.get(
            "content-type",
            "",
        )
    ).casefold()

    if "image/" not in content_type:
        raise RuntimeError(
            "Dataforsyningen returnerede ikke et billede. "
            f"Content-Type var {content_type!r}."
        )

    try:
        with Image.open(
            BytesIO(response.content)
        ) as source_image:
            source_image.load()

            return source_image.convert(
                "RGBA"
            ).copy()

    except Exception as exc:
        raise RuntimeError(
            "Kortsvaret kunne ikke åbnes som et billede."
        ) from exc

def opret_unikke_outputstier(

def gem_kortfiler(
    image: Image.Image,
    points: list[tuple[float, float]],
    bbox: tuple[float, float, float, float],
    *,
    output_mappe: str | Path = OUTPUT_MAPPE,
) -> tuple[Path, Path]:
    """
    Tegner polygonen og gemmer unikke PNG- og PDF-filer.

    Output:
        Tuple med den gemte PNG-sti og PDF-sti.
    """

    if image is None:
        raise ValueError(
            "image skal angives."
        )

    pixel_points = [
        world_to_pixel(
            x=x,
            y=y,
            bbox=bbox,
            width=image.width,
            height=image.height,
        )
        for x, y in points
    ]

    output_image = image.copy()

    draw = ImageDraw.Draw(
        output_image
    )

    draw.line(
        pixel_points,
        fill="red",
        width=POLYGON_LINE_WIDTH,
        joint="curve",
    )

    png_sti, pdf_sti = opret_unikke_outputstier(
        output_mappe
    )

    output_image.save(
        png_sti,
        format="PNG",
    )

    output_image.convert(
        "RGB"
    ).save(
        pdf_sti,
        format="PDF",
        resolution=PDF_RESOLUTION,
    )

    if not png_sti.exists() or png_sti.stat().st_size == 0:
        raise RuntimeError(
            "PNG-filen blev ikke gemt korrekt."
        )

    if not pdf_sti.exists() or pdf_sti.stat().st_size == 0:
        raise RuntimeError(
            "PDF-filen blev ikke gemt korrekt."
        )

    if not pdf_sti.read_bytes().startswith(
        b"%PDF-"
    ):
        raise RuntimeError(
            "Den gemte PDF mangler en gyldig PDF-signatur."
        )

    return (
        png_sti,
        pdf_sti,
    )
